                   "Source-declaration order. Empty description and empty inputs render as em-dash.",

               RunE: func(cmd *cobra.Command, args []string) error {
                   file := args[0]

                   if len(cfg.exts) > 0 {

       // renderInfoTable writes a 3-column table (Flow / Description / Inputs)

       func renderInfoTable(out io.Writer, flows []*dag.Flow) {
           tw := tabwriter.NewWriter(out, 0, 0, 2, ' ', 0)
           fmt.Fprintln(tw, "Flow\tDescription\tInputs")
           for _, f := range flows {
               desc := f.Description
               if desc == "" {
                   desc = emDash
               }
               fmt.Fprintf(tw, "%s\t%s\t%s\n", f.Name, desc, formatInputs(f.Inputs))
           }
           _ = tw.Flush()
       }

       // formatInputs renders an Inputs map as "k:type, k:type" with keys
       // sorted alphabetically. Empty / nil → em-dash.
       func formatInputs(inputs map[string]string) string {
           if len(inputs) == 0 {
               return emDash
           }
           keys := make([]string, 0, len(inputs))
           for k := range inputs {
               keys = append(keys, k)
           }
           sort.Strings(keys)
           var b strings.Builder
           for i, k := range keys {
               if i > 0 {
                   b.WriteString(", ")
               }
               b.WriteString(k)
               b.WriteByte(':')
               b.WriteString(inputs[k])
           }
           return b.String()
       }
