```go
// logStepJSON is the wire shape for *LogStep — mirrors failJSON.
// Pos is intentionally omitted (Filename is machine-absolute; breaks cross-machine golden stability per the file-header rule).
// MsgFn pointer is excluded (lambda non-serializable); only msg_lambda_id surfaces.
type logStepJSON struct {
    Kind          string `json:"kind"`
    Level         string `json:"level"`
    Msg           string `json:"msg"`
    MsgLambdaID   string `json:"msg_lambda_id,omitempty"`
    AttrsLambdaID string `json:"attrs_lambda_id,omitempty"`
}

func (n *LogStep) MarshalJSON() ([]byte, error) {
    var msgLambdaID string
    if n.MsgFn != nil {
        msgLambdaID = n.MsgFn.ID
    }
    return json.Marshal(logStepJSON{
        Kind:          "LogStep",
        Level:         n.Level,
        Msg:           n.Msg,
        MsgLambdaID:   msgLambdaID,
        AttrsLambdaID: n.AttrsLambdaID,
    })
}
```

```go
func TestLogStep_MarshalJSON_Literal(t *testing.T) {
    n := &LogStep{Level: "info", Msg: "hello"}
    b, err := json.Marshal(n)
    require.NoError(t, err)
    // Expect: {"kind":"LogStep","level":"info","msg":"hello"}
    require.JSONEq(t, `{"kind":"LogStep","level":"info","msg":"hello"}`, string(b))
}

func TestLogStep_MarshalJSON_WithLambdas(t *testing.T) {
    n := &LogStep{
        Level:         "warn",
        Msg:           "${ctx.x}",
        MsgFn:         &CapturedLambda{ID: "lam_abc"},
        AttrsLambdaID: "lam_xyz",
    }
    b, err := json.Marshal(n)
    require.NoError(t, err)
    require.JSONEq(t,
        `{"kind":"LogStep","level":"warn","msg":"${ctx.x}","msg_lambda_id":"lam_abc","attrs_lambda_id":"lam_xyz"}`,
        string(b))
}

func TestLogStep_MarshalJSON_NoLambdaSerialized(t *testing.T) {
    // Critical: *starlark.Function inside CapturedLambda must NOT leak into JSON.
    // Only the ID string surfaces.
    n := &LogStep{Level: "info", Msg: "hi", MsgFn: &CapturedLambda{ID: "lam_xyz"}}
    b, err := json.Marshal(n)
    require.NoError(t, err)
    require.NotContains(t, string(b), "Function")
    require.NotContains(t, string(b), "*starlark")
    require.Contains(t, string(b), `"msg_lambda_id":"lam_xyz"`)
}

func TestLogStep_MarshalJSON_InSlice(t *testing.T) {
    // Verify slice-of-Node default marshaling dispatches LogStep.MarshalJSON
    // (sibling shape to other dag nodes).
    nodes := []Node{&LogStep{Level: "info", Msg: "a"}, &LogStep{Level: "warn", Msg: "b"}}
    b, err := json.Marshal(nodes)
    require.NoError(t, err)
    require.Contains(t, string(b), `"level":"info"`)
    require.Contains(t, string(b), `"level":"warn"`)
    require.Contains(t, string(b), `"kind":"LogStep"`)
}
```
