```go
// pkg/extension/extension.go
package extension

import "go.starlark.net/starlark"

// Extension is the SDK contract every library-developer-authored extension implements.
// Examples: pkg/examples/http, pkg/examples/github, pkg/examples/slack (Phase 6).
//
// LIFECYCLE: Initialize is called ONCE per parser at extension Register time —
// NOT once per Starlark invocation. The returned starlark.Value is bound as
// the global keyed by Name() in parseTimeGlobals. From the .star author's view:
//
//   github = <returned starlark.Value>           # bound once at parser registration
//   gh = github.endpoint("admin")                 # attribute lookup + call → returns sub-value
//   gh.create_issue(repo="x", title="y")          # attribute lookup + call → returns *ActionRef
//
// The returned value MUST be attribute-bearing (typically *starlarkstruct.Module
// or any *starlark.Value implementing HasAttrs) so the .star author can write
// `github.endpoint(...)` and `github.<op>(...)`. The attribute named "endpoint"
// is a *starlark.Builtin returning a credential-aware sub-Module that carries
// the credential ID; subsequent ops on that sub-Module build *ActionRef intents
// embedding the resolved CredentialID.
type Extension interface {
    // Name returns the Starlark-namespace identifier (e.g. "github").
    // The parser binds this name as a top-level global to the value returned
    // by Initialize.
    Name() string

    // Initialize is called ONCE per parser at Register time (not per Starlark
    // invocation). Returns the attribute-bearing Starlark value (typically
    // *starlarkstruct.Module) that becomes the global keyed by Name().
    //
    // The returned value's attributes MUST include:
    //   - one attribute per operation declared by Operations() (each a
    //     *starlark.Builtin that, when called, returns a *dag.ActionRef)
    //   - additional factory attributes (e.g., "endpoint") that return a
    //     credential-aware sub-Module — the sub-Module's attributes are
    //     *starlark.Builtins closing over the credential ID and producing
    //     *dag.ActionRef instances with CredentialID populated
    //
    // The kwargs parameter is reserved for parser-level Initialize options
    // (Phase 1: usually empty). It is NOT the per-call kwargs of an extension
    // operation — those flow through the *starlark.Builtin attribute call site.
    Initialize(thread *starlark.Thread, kwargs []starlark.Tuple) (starlark.Value, error)

    // Operations returns the per-extension operation map keyed by operation name
    // (e.g. "create_issue"). The Registry consults this at registration time to
    // verify each operation has Idempotent != nil (D-12). The parser uses these
    // specs to validate operation kwargs at the *starlark.Builtin call site.
    Operations() map[string]*OperationSpec
}
```

```go
package extension

import (
    "context"
    "reflect"
)

// OperationFunc runs INSIDE the generic activity (Phase 2). It takes context.Context
// (stdlib), NEVER workflow.Context — Pitfall #4 prevention. Phase 1 defines the type;
// Phase 2 calls it.
type OperationFunc func(ctx context.Context, args any, cred Credential) (output any, err error)

// OperationSpec is the registration record for one operation.
// Idempotent is *bool (NOT bool) so registration can detect "author forgot to set it"
// vs "author explicitly chose false". Per D-12 — required, no default.
type OperationSpec struct {
    Name       string         // operation name as Starlark callers see it (e.g. "create_issue")
    Idempotent *bool          // REQUIRED — Registry rejects nil
    Func       OperationFunc  // the Go implementation
    KwargsType reflect.Type   // a struct type with `star:"name,required"` tags; ParseSchema reflects on it
}

// Helper for constructing *bool literals in extension definitions.
func Ptr[T any](v T) *T { return &v }
```

```go
// pkg/extension/credential.go
package extension

import "fmt"

// Credential is a sealed interface — the unexported isCredential() method prevents
// downstream packages from declaring new kinds. Adding a kind requires editing
// this file (a deliberate API-evolution gate).
type Credential interface {
    ID() string
    isCredential()
}

type BearerCredential struct {
    ID_   string
    Token string
}

func (b *BearerCredential) ID() string { return b.ID_ }
func (b *BearerCredential) String() string {
    return fmt.Sprintf("<credential:bearer:%s>", b.ID_) // NEVER the token
}
func (*BearerCredential) isCredential() {}

type BasicCredential struct {
    ID_      string
    User     string
    Password string
}

func (b *BasicCredential) ID() string { return b.ID_ }
func (b *BasicCredential) String() string {
    return fmt.Sprintf("<credential:basic:%s>", b.ID_) // NEVER user/password
}
func (*BasicCredential) isCredential() {}

type APIKeyCredential struct {
    ID_        string
    Key        string
    HeaderName string
}

func (a *APIKeyCredential) ID() string { return a.ID_ }
func (a *APIKeyCredential) String() string {
    return fmt.Sprintf("<credential:apikey:%s>", a.ID_) // NEVER the key
}
func (*APIKeyCredential) isCredential() {}
```

```go
// pkg/extension/registry.go
package extension

import (
    "errors"
    "fmt"
    "sync"
)

// ErrIdempotentRequired is returned by Register when an operation lacks Idempotent.
var ErrIdempotentRequired = errors.New("Idempotent declaration required")

// Registry is per-parser (D-07: no global state). Constructed by parser.NewParser
// in plan 05; this file owns the data structure + Register validation.
type Registry struct {
    mu  sync.RWMutex
    ext map[string]Extension // by Extension.Name()
}

func NewRegistry() *Registry {
    return &Registry{ext: make(map[string]Extension)}
}

// Register validates the extension and adds it to the registry.
// FAILS LOUDLY (returns error) if any operation has Idempotent == nil — D-12.
func (r *Registry) Register(ext Extension) error {
    name := ext.Name()
    if name == "" {
        return errors.New("extension name must be non-empty")
    }
    ops := ext.Operations()
    for opName, spec := range ops {
        if spec == nil {
            return fmt.Errorf("extension %q operation %q: nil spec", name, opName)
        }
        if spec.Idempotent == nil {
            return fmt.Errorf("extension %q operation %q: %w", name, opName, ErrIdempotentRequired)
        }
        if spec.Func == nil {
            return fmt.Errorf("extension %q operation %q: nil Func", name, opName)
        }
        if spec.KwargsType == nil {
            return fmt.Errorf("extension %q operation %q: nil KwargsType", name, opName)
        }
    }
    r.mu.Lock()
    defer r.mu.Unlock()
    if _, dup := r.ext[name]; dup {
        return fmt.Errorf("extension %q already registered", name)
    }
    r.ext[name] = ext
    return nil
}

func (r *Registry) Get(name string) (Extension, bool) {
    r.mu.RLock()
    defer r.mu.RUnlock()
    e, ok := r.ext[name]
    return e, ok
}

func (r *Registry) All() map[string]Extension {
    r.mu.RLock()
    defer r.mu.RUnlock()
    out := make(map[string]Extension, len(r.ext))
    for k, v := range r.ext {
        out[k] = v
    }
    return out
}
```

    ```go
    func ParseSchema(t reflect.Type) ([]FieldSpec, error) {
        if t.Kind() == reflect.Ptr {
            t = t.Elem()
        }
        if t.Kind() != reflect.Struct {
            return nil, fmt.Errorf("ParseSchema: %v is not a struct", t)
        }
        var specs []FieldSpec
        for i := 0; i < t.NumField(); i++ {
            f := t.Field(i)
            tag := f.Tag.Get("star")
            if tag == "" || tag == "-" {
                continue
            }
            parts := strings.Split(tag, ",")
            spec := FieldSpec{
                GoName:   f.Name,
                StarName: parts[0],
                GoType:   f.Type,
                FieldIdx: i,
            }
            if spec.StarName == "" {
                return nil, fmt.Errorf("ParseSchema: field %s has empty star: name", f.Name)
            }
            for _, opt := range parts[1:] {
                if opt == "required" {
                    spec.Required = true
                }
            }
            specs = append(specs, spec)
        }
        return specs, nil
    }

    func UnpackOperationKwargs(opName string, callPos syntax.Position, specs []FieldSpec, kwargs []starlark.Tuple, target any) error {
        seen := make(map[string]starlark.Value, len(kwargs))
        for _, kv := range kwargs {
            keyStr, ok := kv[0].(starlark.String)
            if !ok {
                return &dag.ValidationError{Pos: callPos, Msg: fmt.Sprintf("%s: kwarg key must be string", opName)}
            }
            seen[string(keyStr)] = kv[1]
        }
        // Required check
        for _, spec := range specs {
            if spec.Required {
                if _, ok := seen[spec.StarName]; !ok {
                    return &dag.ValidationError{
                        Pos: callPos,
                        Msg: fmt.Sprintf("%s: missing required kwarg %q", opName, spec.StarName),
                    }
                }
            }
        }
        // Unknown check
        known := make(map[string]bool, len(specs))
        for _, s := range specs {
            known[s.StarName] = true
        }
        for k := range seen {
            if !known[k] {
                return &dag.ValidationError{
                    Pos: callPos,
                    Msg: fmt.Sprintf("%s: unknown kwarg %q", opName, k),
                }
            }
        }
        // Populate
        tv := reflect.ValueOf(target).Elem()
        for _, spec := range specs {
            v, present := seen[spec.StarName]
            if !present {
                continue
            }
            if err := assignStarlarkToGo(tv.Field(spec.FieldIdx), v); err != nil {
                return &dag.ValidationError{
                    Pos: callPos,
                    Msg: fmt.Sprintf("%s: kwarg %q: %v", opName, spec.StarName, err),
                }
            }
        }
        return nil
    }

    func assignStarlarkToGo(dst reflect.Value, src starlark.Value) error {
        // None on optional → leave zero value
        if src == starlark.None {
            return nil
        }
        switch dst.Kind() {
        case reflect.String:
            s, ok := src.(starlark.String)
            if !ok {
                return fmt.Errorf("expected string, got %s", src.Type())
            }
            dst.SetString(string(s))
        case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
            i, ok := src.(starlark.Int)
            if !ok {
                return fmt.Errorf("expected int, got %s", src.Type())
            }
            v, ok := i.Int64()
            if !ok {
                return fmt.Errorf("int overflow")
            }
            dst.SetInt(v)
        case reflect.Bool:
            b, ok := src.(starlark.Bool)
            if !ok {
                return fmt.Errorf("expected bool, got %s", src.Type())
            }
            dst.SetBool(bool(b))
        case reflect.Float64:
            f, ok := src.(starlark.Float)
            if !ok {
                return fmt.Errorf("expected float, got %s", src.Type())
            }
            dst.SetFloat(float64(f))
        case reflect.Slice:
            // []string only in Phase 1
            if dst.Type().Elem().Kind() != reflect.String {
                return fmt.Errorf("unsupported slice element type %s", dst.Type().Elem())
            }
            lst, ok := src.(*starlark.List)
            if !ok {
                return fmt.Errorf("expected list, got %s", src.Type())
            }
            out := reflect.MakeSlice(dst.Type(), 0, lst.Len())
            iter := lst.Iterate()
            defer iter.Done()
            var x starlark.Value
            for iter.Next(&x) {
                s, ok := x.(starlark.String)
                if !ok {
                    return fmt.Errorf("expected string in list, got %s", x.Type())
                }
                out = reflect.Append(out, reflect.ValueOf(string(s)))
            }
            dst.Set(out)
        case reflect.Map:
            // map[string]string only in Phase 1
            if dst.Type().Key().Kind() != reflect.String || dst.Type().Elem().Kind() != reflect.String {
                return fmt.Errorf("unsupported map type %s", dst.Type())
            }
            d, ok := src.(*starlark.Dict)
            if !ok {
                return fmt.Errorf("expected dict, got %s", src.Type())
            }
            out := reflect.MakeMapWithSize(dst.Type(), d.Len())
            for _, item := range d.Items() {
                k, ok := item[0].(starlark.String)
                if !ok {
                    return fmt.Errorf("expected string key, got %s", item[0].Type())
                }
                v, ok := item[1].(starlark.String)
                if !ok {
                    return fmt.Errorf("expected string value, got %s", item[1].Type())
                }
                out.SetMapIndex(reflect.ValueOf(string(k)), reflect.ValueOf(string(v)))
            }
            dst.Set(out)
        default:
            return fmt.Errorf("unsupported field type %s (extend assignStarlarkToGo)", dst.Kind())
        }
        return nil
    }
    ```
