```go
package interpreter

import (
    "context"
    "errors"
    "fmt"
    "sync"

    "github.com/stretchr/testify/mock"

    sdkactivity "go.temporal.io/sdk/activity"
    "go.temporal.io/sdk/converter"
    "go.temporal.io/sdk/log"
    "go.temporal.io/sdk/testsuite"
    "go.temporal.io/sdk/workflow"

    "github.com/mikelalcon/skytime/pkg/dag"
)

// EventCapture is the public capture buffer for replay-determinism
// diffing (Phase 5 D5-D1/D5-D4). It records:
//
//   - Interpreter slog events emitted via workflow.GetLogger (the same
//     events that drive the live progress block: flow_start,
//     step_dispatch, step_complete, branch, result_bound,
//     flow_complete).
//   - Activity-boundary events captured via TestWorkflowEnvironment's
//     SetOnActivityStartedListener / SetOnActivityCompletedListener
//     (Investigation 2 — TestWorkflowEnvironment has NO
//     GetWorkflowHistory).
//
// Snapshot returns a stable copy; Serialize returns a byte-string for
// diff. Snapshot+Serialize are concurrency-safe under -race.
type EventCapture struct {
    mu      sync.Mutex
    records []EventRecord
}

// EventRecord is one captured slog or activity-boundary entry. Kind is
// one of "slog", "activity_started", "activity_completed".
type EventRecord struct {
    Kind    string
    Level   string
    Message string
    KV      []any              // slog key-value pairs (sorted before serialize)
    Args    []*dag.ActionRef   // populated for activity_started
    Results []dag.ActionResult // populated for activity_completed
    Err     string             // populated for activity_completed
}

// NewEventCapture returns a fresh capture buffer.
func NewEventCapture() *EventCapture {
    return &EventCapture{}
}

// Implement the temporal log.Logger interface so testsuite.SetLogger
// can route interpreter slog events into this capture.
func (c *EventCapture) Debug(msg string, kv ...any) { c.append("slog", "DEBUG", msg, kv) }
func (c *EventCapture) Info(msg string, kv ...any)  { c.append("slog", "INFO", msg, kv) }
func (c *EventCapture) Warn(msg string, kv ...any)  { c.append("slog", "WARN", msg, kv) }
func (c *EventCapture) Error(msg string, kv ...any) { c.append("slog", "ERROR", msg, kv) }

func (c *EventCapture) append(kind, level, msg string, kv []any) {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.records = append(c.records, EventRecord{Kind: kind, Level: level, Message: msg, KV: kv})
}

// onActivityStarted is the SetOnActivityStartedListener callback.
func (c *EventCapture) onActivityStarted(_ *sdkactivity.Info, _ context.Context, args converter.EncodedValues) {
    var batch []*dag.ActionRef
    if err := args.Get(&batch); err != nil {
        c.mu.Lock()
        defer c.mu.Unlock()
        c.records = append(c.records, EventRecord{
            Kind:    "activity_started",
            Message: fmt.Sprintf("activity_started decode error: %v", err),
        })
        return
    }
    c.mu.Lock()
    defer c.mu.Unlock()
    c.records = append(c.records, EventRecord{
        Kind: "activity_started",
        Args: batch,
    })
}

// onActivityCompleted is the SetOnActivityCompletedListener callback.
func (c *EventCapture) onActivityCompleted(_ *sdkactivity.Info, result converter.EncodedValue, err error) {
    var results dag.ActionResults
    if result != nil {
        _ = result.Get(&results)
    }
    errStr := ""
    if err != nil {
        errStr = err.Error()
    }
    c.mu.Lock()
    defer c.mu.Unlock()
    c.records = append(c.records, EventRecord{
        Kind:    "activity_completed",
        Results: []dag.ActionResult(results),
        Err:     errStr,
    })
}

// Snapshot returns a copy of the records slice.
func (c *EventCapture) Snapshot() []EventRecord {
    c.mu.Lock()
    defer c.mu.Unlock()
    out := make([]EventRecord, len(c.records))
    copy(out, c.records)
    return out
}

// Serialize renders the capture as a stable byte string for diff. Uses
// a deterministic format: one line per record, KV pairs sorted by key,
// activity records emit a JSON-stable shape via dag's MarshalJSON.
func (c *EventCapture) Serialize() []byte {
    return serializeEventCapture(c)
}

// Compile-time interface assertion: EventCapture satisfies log.Logger.
var _ log.Logger = (*EventCapture)(nil)

// RunOnceCapturing executes parsed against a fresh
// TestWorkflowEnvironment, capturing slog events + activity-boundary
// events. Pass mockCallback == nil for if_cond/script-only flows that
// emit no activities. Phase 5's pkg/testing wraps this to thread
// in the buildExecuteBatchCallback router.
//
// Replaces the file-private runOnceCapturing in
// replay_determinism_test.go (Phase 04.2). Existing internal tests
// migrate to call this public name.
func RunOnceCapturing(
    parsed *ParsedFlow,
    hash string,
    init map[string]any,
    mockCallback func(context.Context, []*dag.ActionRef) ([]dag.ActionResult, error),
) (*EventCapture, map[string]any, error) {
    if parsed == nil {
        return nil, nil, errors.New("RunOnceCapturing: parsed must not be nil")
    }

    cap := NewEventCapture()
    registry := NewRegistry()
    if err := registry.Register(parsed.Flow.Name, hash, parsed); err != nil {
        return nil, nil, fmt.Errorf("registry.Register: %w", err)
    }
    registry.Freeze()

    var ts testsuite.WorkflowTestSuite
    ts.SetLogger(cap)
    env := ts.NewTestWorkflowEnvironment()
    env.SetOnActivityStartedListener(cap.onActivityStarted)
    env.SetOnActivityCompletedListener(cap.onActivityCompleted)

    if mockCallback != nil {
        fake := func(_ context.Context, _ []*dag.ActionRef) ([]dag.ActionResult, error) { return nil, nil }
        env.RegisterActivityWithOptions(fake, sdkactivity.RegisterOptions{Name: "ExecuteBatch"})
        env.OnActivity("ExecuteBatch", mock.Anything, mock.Anything).Return(mockCallback)
    }

    wf := NewWorkflow(registry)
    env.RegisterWorkflowWithOptions(wf, workflow.RegisterOptions{Name: "SkytimeWorkflow"})
    env.ExecuteWorkflow(wf, dag.WorkflowInput{
        FlowName:    parsed.Flow.Name,
        ContentHash: hash,
        InitState:   init,
    })
    if !env.IsWorkflowCompleted() {
        return cap, nil, errors.New("workflow did not complete")
    }
    var out map[string]any
    wfErr := env.GetWorkflowError()
    if wfErr == nil {
        _ = env.GetWorkflowResult(&out)
    }
    return cap, out, wfErr
}
```

```go
package testing

import (
```

```go
    "fmt"

    "go.starlark.net/syntax"

    "github.com/mikelalcon/skytime/pkg/interpreter"

// Divergence is the D5-D2 first-divergent-event report. Returned by
// FirstDivergentEvent; nil if the two captures serialize byte-equal.
//
// FlowCallsite is the originating step's syntax.Position read from the
// nearest preceding step_dispatch event's `pos` KV (added by Plan 03
// Task 0); empty for events that occur before any step_dispatch.
// TestCallsite is the caller-supplied position (typically the .star
// file's tester.run(...) line).
type Divergence struct {
    Index        int
    Kind         string
    Before       string
    After        string
    FlowCallsite syntax.Position
    StepName     string
    TestCallsite syntax.Position
}

// FirstDivergentEvent returns the first record index where run1 and
// run2 disagree. Returns nil if both captures serialize byte-equal.
//
// The diff scope (D5-D4) is the EXISTING serialization shape:
// EventCapture.Serialize emits one line per record; we walk
// records line-by-line until the first inequality, then attribute
// the flow callsite via the matching step_dispatch event upstream.
func FirstDivergentEvent(run1, run2 *interpreter.EventCapture, testCallsite syntax.Position) *Divergence {
    snap1 := run1.Snapshot()
    snap2 := run2.Snapshot()

    n := len(snap1)
    if len(snap2) < n {
        n = len(snap2)
    }

    for i := 0; i < n; i++ {
        a := serializeOne(snap1[i])
        b := serializeOne(snap2[i])
        if a != b {
            d := &Divergence{
                Index:        i,
                Kind:         snap1[i].Kind,
                Before:       a,
                After:        b,
                TestCallsite: testCallsite,
            }
            if pos, name, ok := lookupOriginatingStep(snap1, i); ok {
                d.FlowCallsite = pos
                d.StepName = name
            }
            return d
        }
    }

    // Structural divergence: lengths differ.
    if len(snap1) != len(snap2) {
        idx := n
        var before, after string
        if idx < len(snap1) {
            before = serializeOne(snap1[idx])
        } else {
            before = "<missing>"
        }
        if idx < len(snap2) {
            after = serializeOne(snap2[idx])
        } else {
            after = "<missing>"
        }
        d := &Divergence{
            Index:        idx,
            Kind:         "<structural>",
            Before:       before,
            After:        after,
            TestCallsite: testCallsite,
        }
        // Attribute via run1 if possible.
        if pos, name, ok := lookupOriginatingStep(snap1, idx); ok {
            d.FlowCallsite = pos
            d.StepName = name
        }
        return d
    }

    return nil
}

// serializeOne renders one EventRecord to a stable line. Mirrors
// interpreter.EventCapture.Serialize's per-record format so the diff
// granularity is identical.
func serializeOne(r interpreter.EventRecord) string {
    var buf bytes.Buffer
    fmt.Fprintf(&buf, "%s|%s|%s", r.Kind, r.Level, r.Message)
    // KV pairs already sorted by interpreter.serializeEventCapture; if
    // not, sort here. Defensive sort to keep the diff helper
    // self-contained.
    if len(r.KV) > 0 {
        // Render KV as |key=value|key=value... in stable order.
        // (Implementation detail: when porting from
        // interpreter.serializeEventCapture, reuse the same sort+render
        // pattern. ~10 LOC.)
    }
    return buf.String()
}

// lookupOriginatingStep walks backward from `idx` looking for the
// nearest step_dispatch event; extracts `pos` and `name` attributes
// (added by Plan 03 Task 0 to walk_step.go). Returns (zero, "", false)
// if no step_dispatch event precedes idx (e.g., a divergence at
// flow_start).
func lookupOriginatingStep(snap []interpreter.EventRecord, idx int) (syntax.Position, string, bool) {
    for i := idx; i >= 0; i-- {
        r := snap[i]
        // Interpreter emits "skytime" as the slog message and "step_dispatch"
        // under the "event" KV. Match on the KV pair, not the message.
        if r.Kind != "slog" {
            continue
        }
        if !kvEquals(r.KV, "event", "step_dispatch") {
            continue
        }
        pos, posOK := extractKV[syntax.Position](r.KV, "pos")
        name, _ := extractKV[string](r.KV, "name")
        if posOK {
            return pos, name, true
        }
    }
    return syntax.Position{}, "", false
}

// kvEquals returns true when kv contains the (key, value) pair.
func kvEquals(kv []any, key string, value any) bool {
    for i := 0; i+1 < len(kv); i += 2 {
        k, ok := kv[i].(string)
        if !ok || k != key {
            continue
        }
        return kv[i+1] == value
    }
    return false
}

func extractKV[T any](kv []any, key string) (T, bool) {
    var zero T
    for i := 0; i+1 < len(kv); i += 2 {
        k, ok := kv[i].(string)
        if !ok || k != key {
            continue
        }
        if v, ok := kv[i+1].(T); ok {
            return v, true
        }
    }
    return zero, false
}

// Format renders the D5-D2 multi-line message:
//
//   replay diverged
//     event N (KIND) diverged:
//       run1: <serialized>
//       run2: <serialized>
//     flow callsite: <flow_file>:<line>:<col> (step "<name>")
//     test callsite: <test_file>:<line>:<col> (tester.run)
func (d *Divergence) Format() string {
    var buf bytes.Buffer
    fmt.Fprintln(&buf, "replay diverged")
    fmt.Fprintf(&buf, "  event %d (%s) diverged:\n", d.Index, d.Kind)
    fmt.Fprintf(&buf, "    run1: %s\n", d.Before)
    fmt.Fprintf(&buf, "    run2: %s\n", d.After)
    if d.FlowCallsite.Line > 0 {
        fmt.Fprintf(&buf, "  flow callsite: %s (step %q)\n", d.FlowCallsite.String(), d.StepName)
    }
    if d.TestCallsite.Line > 0 {
        fmt.Fprintf(&buf, "  test callsite: %s (tester.run)\n", d.TestCallsite.String())
    }
    return buf.String()
}
```
