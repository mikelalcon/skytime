    Step 1 — Replace walk.go stubs with real implementations.

    WalkRegistry(path string) (map[string]string, []string, error):
    1. token.NewFileSet(); parser.ParseFile(fset, path, nil, parser.ParseComments). On error, return (nil, nil, err) — error already includes path.
    2. registry := map[string]string{}; var order []string.
    3. ast.Inspect over file. When visiting a *ast.FuncDecl, check funcDecl.Name.Name == "newParseTimeGlobals"; if not, return true (keep walking).
    4. Inside the matched func body, ast.Inspect for *ast.CompositeLit. Check the Type is *ast.SelectorExpr with X.(*ast.Ident).Name == "starlark" and Sel.Name == "StringDict". If not, return true.
    5. Iterate comp.Elts. Each entry should be *ast.KeyValueExpr; skip non-matching.
    6. Key extraction: kv.Key must be *ast.BasicLit with Kind == token.STRING. Use strconv.Unquote to get the raw name string.
    7. Value extraction: kv.Value must be *ast.CallExpr. The CallExpr.Fun must be *ast.SelectorExpr with X.(*ast.Ident).Name == "starlark" and Sel.Name == "NewBuiltin". Skip if mismatch.
    8. Args[1] must be *ast.SelectorExpr (e.g. p.builtinFlow). Function name = sel.Sel.Name.
    9. registry[name] = funcName; order = append(order, name). Return false from the inner ast.Inspect to avoid descending further once we've processed the StringDict; return false from outer to stop after the first newParseTimeGlobals.
    10. Return (registry, order, nil) — empty map + empty order if function not found is also valid.

    WalkBuiltins(path string, registry map[string]string, order []string) ([]Builtin, error):
    1. token.NewFileSet(); parser.ParseFile(fset, path, nil, parser.ParseComments). On error, return nil, err.
    2. byFunc := map[string]string{}; for k,v := range registry { byFunc[v] = k } — reverse map.
    3. found := map[string]Builtin{}.

    ```
    package main

    import (
        "fmt"
        "go/ast"
        "os"
        "strconv"
        "strings"
    )

    const markerPrefix = "// skytime:doc "

    // ParseMarkers extracts `// skytime:doc key="value"` markers from
    // decl's leading CommentGroup. Multi-line values: repeated keys
    // accumulate into the slice. Malformed lines write a warning to
    // stderr and are skipped — never abort.
    //
    // Returns an empty map when no markers are present (NEVER nil for
    // present-but-empty cases; nil is acceptable when decl.Doc is nil).
    func ParseMarkers(decl *ast.FuncDecl) map[string][]string {
        if decl == nil || decl.Doc == nil { return map[string][]string{} }
        out := map[string][]string{}
        for _, c := range decl.Doc.List {
            // c.Text includes the leading "//" (and possibly trailing newline-stripped). Normalize.
            line := c.Text
            // Single-line `// foo` style only — block comments unsupported.
            if !strings.HasPrefix(line, "// skytime:doc ") {
                continue
            }
            payload := strings.TrimPrefix(line, "// skytime:doc ")
            payload = strings.TrimSpace(payload)
            if payload == "" {
                fmt.Fprintf(os.Stderr, "skytime-docgen: warning: empty skytime:doc marker at %s\n", positionString(decl, c))
                continue
            }
            // Split on first '=' — keys are bare identifiers, values are double-quoted Go-style strings.
            eq := strings.Index(payload, "=")
            if eq < 0 {
                fmt.Fprintf(os.Stderr, "skytime:doc warning: missing '=' in marker %q at %s\n", payload, positionString(decl, c))
                continue
            }
            key := strings.TrimSpace(payload[:eq])
            rawValue := strings.TrimSpace(payload[eq+1:])
            if !strings.HasPrefix(rawValue, "\"") {
                fmt.Fprintf(os.Stderr, "skytime:doc warning: value must be double-quoted: %q at %s\n", payload, positionString(decl, c))
                continue
            }
            // strconv.Unquote handles backslash escapes including \" and \\.
            value, err := strconv.Unquote(rawValue)
            if err != nil {
                fmt.Fprintf(os.Stderr, "skytime:doc warning: cannot unquote %q at %s: %v\n", rawValue, positionString(decl, c), err)
                continue
            }
            if key == "" {
                fmt.Fprintf(os.Stderr, "skytime:doc warning: empty key in marker %q at %s\n", payload, positionString(decl, c))
                continue
            }
            out[key] = append(out[key], value)
        }
        return out
    }

    // positionString returns a best-effort source position string for
    // diagnostic output. The caller passes the FuncDecl whose Doc the
    // comment belongs to plus the comment itself; we approximate the
    // line via comment position offset relative to the FuncDecl start.
    func positionString(decl *ast.FuncDecl, c *ast.Comment) string {
        return fmt.Sprintf("%s (offset %d)", decl.Name.Name, c.Slash)
    }
    ```
