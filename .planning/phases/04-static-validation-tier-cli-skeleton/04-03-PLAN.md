Pitfall #7 mitigation: also assert no internal panics on either side.

The differential test is the *enforcement* — drift fails CI. The dispatch helper is what makes the dry-run achievable without standing up real backends.

       package validator

       import (
           "github.com/mikelalcon/skytime/pkg/extension"
       )

       // Option configures a Validate call. Mirrors pkg/parser's option style.
       type Option func(*config) error

       type config struct {
           root        string
           exts        []extension.Extension
           credHandler extension.CredentialHandler
       }

       // WithExtensions registers extensions with the underlying parser.
       func WithExtensions(exts ...extension.Extension) Option {
           return func(c *config) error {
               c.exts = append(c.exts, exts...)
               return nil
           }
       }

       // WithCredentialHandler is accepted for API symmetry with pkg/cli's
       // root command — the validator does not invoke the resolver itself
       // (validate is parse-only), but pkg/cli plumbs the same option down
       // to both validator.Validate and worker.NewWorker.
       func WithCredentialHandler(h extension.CredentialHandler) Option {
           return func(c *config) error {
               c.credHandler = h
               return nil
           }
       }

       // WithRoot sets the parser's load() sandbox root. Defaults to the
       // file's directory when not set; pkg/cli passes this when --rootdir
       // is supplied.
       func WithRoot(root string) Option {
           return func(c *config) error {
               c.root = root
               return nil
           }

       // Package validator is the thin façade in front of pkg/parser for
       // the skytime validate CLI subcommand and the differential-corpus
       // dry-run test (D4-01, D4-03 — Phase 4 CONTEXT.md).
       //
       // The validator does NOT implement new checks — every check (kwarg
       // cross-validate, ctx.<name> walk, free-vars-reference-state, lambda
       // global subset) lives in pkg/parser/finalize.go. The facade exists
       // to give the CLI a single non-cobra entry point and to host the
       // differential-corpus test.
       package validator

       import (
           "github.com/mikelalcon/skytime/pkg/parser"
       )

       // Validate parses file through pkg/parser and returns the typed errors
       // (*dag.ParseError or *dag.ValidationError) as a slice.
       //
       // Returns:
       //   - empty []error on success (NOT nil — the empty slice signals "no
       //     errors" unambiguously)
       //   - one-element slice on parser construction failure (e.g., extension
       //     ErrIdempotentRequired)
       //   - one-element slice on parse failure (the parser short-circuits on
       //     first error per finalize-pass convention)
       //
       // Future enhancement: collect-then-return-all errors when the parser
       // grows multi-error reporting. For v1, single-error short-circuit
       // matches the runtime parser's behavior 1:1, which is exactly what
       // VAL-02's "static + runtime agree" demands.
       func Validate(file string, opts ...Option) []error {
           cfg := &config{}
           for _, opt := range opts {
               if err := opt(cfg); err != nil {
                   return []error{err}
               }
           }

           parserOpts := []parser.Option{}
           if cfg.root != "" {
               parserOpts = append(parserOpts, parser.WithRoot(cfg.root))
           }
           for _, e := range cfg.exts {
               parserOpts = append(parserOpts, parser.WithExtensions(e))
           }

           p, err := parser.NewParser(parserOpts...)
           if err != nil {
               return []error{err}
           }
           if _, err := p.ParseFile(file); err != nil {
               return []error{err}
           }
           return []error{} // no errors — empty (non-nil) slice
       }
       ```

       package dryrun

       import (
           "context"

           "github.com/mikelalcon/skytime/pkg/activity"
           "github.com/mikelalcon/skytime/pkg/dag"
           "github.com/mikelalcon/skytime/pkg/extension"
       )

       // AlwaysOkDispatch wraps each registered op's Func with one that
       // returns (nil, nil) — the activity layer encodes that as OkResult{}.
       //
       // The wrap preserves Name, Idempotent, KwargsType, and DefaultTimeout
       // so the activity's own kwarg-shape checks (DecodeKwargsFromDict) still
       // fire on bad inputs. The only thing replaced is the I/O — the dry-run
       // never makes a real network/filesystem call.
       //
       // Used by pkg/validator/dryrun_test.go's TestDifferentialCorpus to
       // run every .star file in examples/skeleton/ through the full
       // SkytimeWorkflow without standing up real backends.
       func AlwaysOkDispatch(exts []extension.Extension) activity.OperationDispatch {
           d := activity.OperationDispatch{}
           for _, e := range exts {
               extName := e.Name()
               for opName, spec := range e.Operations() {
                   if spec == nil {
                       continue
                   }
                   wrapped := *spec // shallow copy — KwargsType/Idempotent are pointers, OK to share
                   wrapped.Func = okFunc
                   d[extName+"."+opName] = wrapped
               }
           }
           return d
       }

       // okFunc is the always-OK OperationFunc. Returns nil typed Output (legal
       // per pkg/extension/operation.go doc) and nil error → activity emits
       // OkResult{Output: nil} which the interpreter treats as a clean success.
       func okFunc(ctx context.Context, args any, cred extension.Credential) (dag.OperationOutput, error) {
           return nil, nil
       }
       ```

       func TestDifferentialCorpus(t *testing.T) {
           moduleRoot := findValidatorModuleRoot(t)
           corpusDir := filepath.Join(moduleRoot, "examples", "skeleton")

           if _, err := os.Stat(corpusDir); errors.Is(err, fs.ErrNotExist) {
               t.Skipf("corpus dir %s does not exist yet — W4 will populate it", corpusDir)
               return
           }

           var starFiles []string
           require.NoError(t, filepath.WalkDir(corpusDir, func(path string, d fs.DirEntry, walkErr error) error {
               if walkErr != nil {
                   return walkErr
               }
               if !d.IsDir() && filepath.Ext(path) == ".star" {
                   starFiles = append(starFiles, path)
               }
               return nil
           }))

               return

               t.Run(filepath.Base(file), func(t *testing.T) {

                   staticErrs := validator.Validate(file,
                       validator.WithExtensions(exts...),
                       validator.WithRoot(corpusDir),
                   )
                   staticPassed := len(staticErrs) == 0

                   // Dry-run side: parse → build registry → run SkytimeWorkflow

                   dryRunPassed, dryRunErr := runDryRun(t, file, corpusDir, exts)

                   // runtime panic. Both must be user-error class
                   // (*dag.ParseError, *dag.ValidationError, or wrapped
                   // application errors from Temporal).
                   if !staticPassed && !dryRunPassed {
                       for _, e := range staticErrs {

                       assertNotRuntimePanic(t, "dryrun", dryRunErr)

       // runDryRun parses the file, builds a FlowRegistry, and runs
       // SkytimeWorkflow against testsuite.TestWorkflowEnvironment for every
       // declared flow. Returns (passed, firstError). Empty error means all
       // flows ran cleanly. The first failure short-circuits — matches static
       // side's first-error behavior.
       //
       // For the corpus this iterates every flow in the file because we don't
       // know the consultant's intended entry flow at validation time. A
       // single flow failing fails the dry-run side.
       func runDryRun(t *testing.T, file, rootDir string, exts []extension.Extension) (bool, error) {
           t.Helper()

           // Parse with the same options the worker uses.
           parserOpts := []parser.Option{parser.WithRoot(rootDir)}
           for _, e := range exts {
               parserOpts = append(parserOpts, parser.WithExtensions(e))
           }
           p, err := parser.NewParser(parserOpts...)
           if err != nil {
               return false, err
           }
           flows, err := p.ParseFile(file)
           if err != nil {
               return false, err
           }

           // Build a FlowRegistry. We compute a synthetic content_hash since
           // the test does not need cross-run hash stability — boot uses the
           // file path + filebytes hash; here we just need consistent keys.
           reg := interpreter.NewRegistry()
           lambdas := p.Lambdas()
           for name, f := range flows {
               // Use the captured fileBytes hash; bootRegistry's algorithm
               // is sha256 of file bytes — replicate using the parser's
               // FileBytes accessor.
               contentHash := sha256Hex(p.FileBytes()[f.Pos.Filename()])
               require.NoError(t, reg.Register(name, contentHash, &interpreter.ParsedFlow{
                   Flow:    f,
                   Lambdas: lambdas,
               }))
               _ = name
           }
           reg.Freeze()

           // Build the activity with AlwaysOkDispatch (no real I/O).
           dispatch := dryrun.AlwaysOkDispatch(exts)
           act, err := activity.New(dispatch, noopCredentialHandler{})
           if err != nil {
               return false, err
           }

           // Run every flow through TestWorkflowEnvironment. First failing
           // flow stops the loop — same first-error behavior as the static
           // side.
           ts := &testsuite.WorkflowTestSuite{}
           for name := range flows {
               env := ts.NewTestWorkflowEnvironment()
               env.RegisterWorkflowWithOptions(interpreter.NewWorkflow(reg), workflow.RegisterOptions{Name: "SkytimeWorkflow"})
               env.RegisterActivityWithOptions(act.ExecuteBatch, activity.RegisterOptions{Name: "ExecuteBatch"})

               contentHash, ok := reg.ContentHashFor(name)
               require.True(t, ok)
               env.ExecuteWorkflow("SkytimeWorkflow", dag.WorkflowInput{
                   FlowName:    name,
                   ContentHash: contentHash,
                   InitState:   map[string]any{},
               })
               if !env.IsWorkflowCompleted() {
                   return false, errors.New("workflow did not complete")
               }
               if werr := env.GetWorkflowError(); werr != nil {
                   return false, werr
               }
           }
           return true, nil
       }

       // assertNotRuntimePanic checks that err's chain does not contain a
       // runtime.Error (Pitfall #7 in 04-RESEARCH.md). User-error classes
       // (*dag.ParseError, *dag.ValidationError, Temporal application errors)
       // are acceptable; nil-deref / index-out-of-bounds / etc. are bugs.
       func assertNotRuntimePanic(t *testing.T, side string, err error) {
           t.Helper()
           if err == nil {
               return
           }
           var re runtime.Error
           if errors.As(err, &re) {
               t.Errorf("%s side: error chain contains runtime.Error (panic-class), should be user-error: %v", side, err)
           }
       }

       // noopCredentialHandler is the no-op handler used by the dry-run path
       // (the AlwaysOkDispatch never reaches the resolver, but activity.New
       // requires a non-nil handler).
       type noopCredentialHandler struct{}

       func (noopCredentialHandler) Resolve(_ context.Context, _ string) (extension.Credential, error) {
           return nil, errors.New("noopCredentialHandler.Resolve called — dry-run should not require credentials")
       }

       // sha256Hex matches pkg/worker/boot.go's content_hash format.
       func sha256Hex(b []byte) string {
           sum := sha256.Sum256(b)
           return hex.EncodeToString(sum[:])
       }

               }

               if parent == dir {
                   t.Fatalf("could not find go.mod walking up from %s", cwd)
               }
               dir = parent
           }
