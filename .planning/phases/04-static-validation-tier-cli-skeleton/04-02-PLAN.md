<objective>
Wave 1 fills the load-bearing static-validation logic in `pkg/parser/finalize.go`:

1. **D4-02 ctx.<name> attribute walk.** The runtime parser already enforces D-19 (free-vars must reference module-level constants); D4-02 adds the deeper check that every `ctx.<attr>` access inside a captured lambda resolves to a state name visible at the lambda's source position. The state schema accumulates per body walk: flow inputs at entry, += `script.OutputAlias`, += `for_each_parallel.ItemVar` inside the fan-out, and `if_cond` branches see the same pre-branch state.

2. **D-11 kwarg cross-validate (defense in depth).** The existing `validateActionRefKwargs` no-op is filled to walk every `dag.ActionRef` in every flow's body and re-run `extension.UnpackOperationKwargs` against the registered `OperationSpec.KwargsType`. This catches hand-built ActionRefs (test fixtures, future programmatic callers) where the per-call extension factory was bypassed.

The implementation is the load-bearing technical risk of Phase 4 per RESEARCH §Pattern 3: `*starlark.Function` does NOT retain its AST after compilation, so the visitor must re-parse `Parser.FileBytes()[filename]` via `(*syntax.FileOptions).Parse` and match each captured lambda by `Position()` against `*syntax.LambdaExpr.Lambda` and `*syntax.DefStmt.Def`.

Purpose: Without D4-02, `skytime validate` cannot reject `ctx.repo_nme` (typo) or `ctx.unknown` (refactor regression) — the runtime parser already accepts these because Starlark's resolver only verifies names at lambda-call time, which never happens at parse. This is the load-bearing VAL-01 check.

       ```go
       // findCtxAccesses re-parses src to recover the AST, locates the lambda
       // (or def) whose keyword position equals lambdaPos, and returns every
       // <firstParam>.<attr> access in its body.
       //
       // Implementation note: *starlark.Function does NOT retain its AST after
       // compilation (RESEARCH critical finding). The captured lambda's Fn
       // pointer is the runtime closure; the AST is reachable only by re-parse
       // of the original source bytes. This function is invoked from the
       // finalize pass with src = parser.FileBytes()[filename], so re-parse is
       // O(file_bytes) and runs once per validate / once per parse — cheap.
       //
       // Position match strategy: walk every *syntax.LambdaExpr and *syntax.DefStmt
       // in the re-parsed File and pick the one whose keyword position equals
       // lambdaPos. starlark-go's *Function.Position() returns the keyword position
       // (verified via pkg/parser/lambda_capture.go's `pos := fn.Position()`),
       // so the match condition is LambdaExpr.Lambda == lambdaPos OR DefStmt.Def
       // == lambdaPos. Two lambdas on the same line are distinguishable because
       // their Col fields differ (Pitfall #1 fixture covers).
       //
       // Returns ([], nil) when no matching lambda is found in the re-parsed
       // file — defensive; callers treat absent lambdas as "nothing to check".
       func findCtxAccesses(src []byte, filename string, lambdaPos syntax.Position) ([]ctxAccess, error) {
           opts := defaultFileOptions()
           file, err := opts.Parse(filename, src, 0)
           if err != nil {
               return nil, err
           }

           var (
               firstParamName string
               targetBody     syntax.Expr
               targetStmts    []syntax.Stmt
           )
           syntax.Walk(file, func(n syntax.Node) bool {
               switch fn := n.(type) {
               case *syntax.LambdaExpr:
                   if positionsEqual(fn.Lambda, lambdaPos) && len(fn.Params) > 0 {
                       firstParamName = paramName(fn.Params[0])
                       targetBody = fn.Body
                   }
               case *syntax.DefStmt:
                   if positionsEqual(fn.Def, lambdaPos) && len(fn.Params) > 0 {
                       firstParamName = paramName(fn.Params[0])
                       targetStmts = fn.Body
                   }
               }
               return true
           })

           if firstParamName == "" {
               return nil, nil
           }

           var accesses []ctxAccess
           collect := func(n syntax.Node) bool {
               if dot, ok := n.(*syntax.DotExpr); ok {
                   if id, ok := dot.X.(*syntax.Ident); ok && id.Name == firstParamName {
                       accesses = append(accesses, ctxAccess{
                           Pos:      dot.NamePos,
                           AttrName: dot.Name.Name, // Pitfall #9: .Name is *Ident, .Name.Name is string
                       })
                   }
               }
               return true
           }
           if targetBody != nil {
               syntax.Walk(targetBody, collect)
           }
           for _, stmt := range targetStmts {
               syntax.Walk(stmt, collect)
           }
           return accesses, nil
       }

       // positionsEqual compares two syntax.Position values on (Filename, Line,
       // Col). The Pos's Filename returned by syntax.Walk on a re-parsed File
       // matches what the parser stored on dag.CapturedLambda.Pos because both
       // re-parse and original parse used the same filename argument.
       func positionsEqual(a, b syntax.Position) bool {
           return a.Filename() == b.Filename() && a.Line == b.Line && a.Col == b.Col
       }

       // paramName extracts the parameter name from a *syntax.Ident wrapped in
       // a parameter Expr. For LambdaExpr/DefStmt, the param shape is one of:
       //   *syntax.Ident                   — plain `x`
       //   *syntax.BinaryExpr (=)          — defaulted `x = expr`
       //   *syntax.UnaryExpr (* / **)      — *args / **kwargs
       // For Phase 4's check we only care about plain Ident or defaulted Ident.
       func paramName(e syntax.Expr) string {
           switch p := e.(type) {
           case *syntax.Ident:
               return p.Name
           case *syntax.BinaryExpr:
               if id, ok := p.X.(*syntax.Ident); ok {
                   return id.Name
               }
           case *syntax.UnaryExpr:
               if id, ok := p.X.(*syntax.Ident); ok {
                   return id.Name
               }
           }
           return ""
       }
       ```

       ```go
       func (s stateSet) add(name string)         { s[name] = struct{}{} }
       func (s stateSet) has(name string) bool    { _, ok := s[name]; return ok }
       func (s stateSet) clone() stateSet {
           out := make(stateSet, len(s))
           for k := range s {
               out[k] = struct{}{}
           }
           return out
       }
       func (s stateSet) sortedKeys() []string {
           keys := make([]string, 0, len(s))
           for k := range s {
               keys = append(keys, k)
           }
           sort.Strings(keys)
           return keys
       }

       // validateLambdaCtxAccesses walks every flow's body sequentially. For
       // each captured lambda it computes the state set visible at the
       // lambda's source position and rejects any ctx.<name> reference that
       // is not in the set.
       //
       // D4-02 stacking rules:
       //   - At flow entry: state = keys of flow.Inputs
       //   - Script: validate lambda BEFORE adding OutputAlias; then state += OutputAlias
       //   - IfCond: validate cond lambda; walk Then and Else each with a CLONE of pre-branch state
       //   - ForEachParallel: validate items lambda (if ItemsLambdaID set) against pre-loop state;
       //     then walk Steps with a clone += ItemVar
       //   - Step: no lambda evaluation (Phase 4 does not check kwarg expressions in this pass)
       //   - CallFlow: no lambda evaluation (D-19 forbids lambdas in CallFlow inputs)
       func (p *Parser) validateLambdaCtxAccesses() error {
           for _, flow := range p.flows {
               initial := newStateSet()
               for k := range flow.Inputs {
                   initial.add(k)
               }
               if err := p.walkBodyForCtxValidation(flow, flow.Body, initial); err != nil {
                   return err
               }
           }
           return nil
       }

       // walkBodyForCtxValidation is the recursive helper. Mirrors
       // walkLintMixedIdempotency / walkLintBlockSize for grep-friendly recursion.
       func (p *Parser) walkBodyForCtxValidation(flow *dag.Flow, body []dag.Node, state stateSet) error {
           for _, node := range body {
               switch n := node.(type) {
               case *dag.Script:
                   if err := p.checkLambdaCtx(flow, n.LambdaID, state); err != nil {
                       return err
                   }
                   if n.OutputAlias != "" {
                       state.add(n.OutputAlias)
                   }
               case *dag.IfCond:
                   if err := p.checkLambdaCtx(flow, n.LambdaID, state); err != nil {
                       return err
                   }
                   if err := p.walkBodyForCtxValidation(flow, n.Then, state.clone()); err != nil {
                       return err
                   }
                   if err := p.walkBodyForCtxValidation(flow, n.Else, state.clone()); err != nil {
                       return err
                   }
               case *dag.ForEachParallel:
                   if n.ItemsLambdaID != "" {
                       // The items producer sees only PRE-loop state.
                       if err := p.checkLambdaCtx(flow, n.ItemsLambdaID, state); err != nil {
                           return err
                       }
                   }
                   inner := state.clone()
                   if n.ItemVar != "" {
                       inner.add(n.ItemVar)
                   }
                   if err := p.walkBodyForCtxValidation(flow, n.Steps, inner); err != nil {
                       return err
                   }
               case *dag.Step, *dag.CallFlow:
                   // No lambda evaluation: Step kwargs are pure data; CallFlow
                   // inputs forbid lambdas (D-19). Nothing to validate here.
                   _ = n
               }
           }
           return nil
       }

       // checkLambdaCtx finds the lambda by ID, reads its source bytes from
       // the parser's fileBytes cache, and emits a *dag.ValidationError on
       // the first ctx.<name> access not in state.
       //
       // Empty lambdaID is silently skipped (defensive — well-formed nodes
       // always carry an ID after lambda capture, but tests may inject
       // partial data).
       func (p *Parser) checkLambdaCtx(flow *dag.Flow, lambdaID string, state stateSet) error {
           if lambdaID == "" {
               return nil
           }
           captured, ok := p.lambdas[lambdaID]
           if !ok {
               return nil
           }
           src, ok := p.fileBytes[captured.Pos.Filename()]
           if !ok || src == nil {
               return nil // bytes not cached → defensive skip; lambda capture would have errored earlier
           }
           accesses, err := findCtxAccesses(src, captured.Pos.Filename(), captured.Pos)
           if err != nil {
               return &dag.ValidationError{
                   Pos:     captured.Pos,
                   Flow:    flow.Name,
                   Msg:     fmt.Sprintf("re-parse for ctx.<name> validation failed: %v", err),
                   Wrapped: err,
               }
           }
           for _, acc := range accesses {
               if !state.has(acc.AttrName) {
                   return &dag.ValidationError{
                       Pos:  acc.Pos,
                       Flow: flow.Name,
                       Msg:  fmt.Sprintf("ctx.%s not in declared state (visible: %v)", acc.AttrName, state.sortedKeys()),
                   }
               }
           }
           return nil
       }
       ```

       ```go
       func (p *Parser) walkValidateActionRefKwargs(flowName string, body []dag.Node) error {

       // surface these conditions with better attribution. The point of this
       // pass is to catch SCHEMA mismatches that slipped through.
       func (p *Parser) crossValidateActionRef(flowName string, stepPos syntax.Position, ar *dag.ActionRef) error {
           extName, opName, ok := splitKind(ar.Kind_)
           if !ok {

           if !ok {
               return nil

           if err := extension.DecodeKwargsFromDict(opName, ar.Kwargs, target); err != nil {

                   Pos:     stepPos,
                   Flow:    flowName,
                   Action:  ar.Kind_,
                   Msg:     fmt.Sprintf("kwarg cross-validate: %v", err),
                   Wrapped: err,
               }
           }

       }
       ```
