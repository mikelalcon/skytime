// pkg/bridge/struct.go
package bridge

import (
    "fmt"
    "sort"

    "go.starlark.net/starlark"
    "go.starlark.net/starlarkstruct"
)

// ToStarlarkStruct recursively converts a Go state map into a *starlarkstruct.Struct
// with deterministic key order (sort.Strings before iteration — Pitfall #3).
//
// Supported value types in v1: string, int (and friends), float64, bool, []any, map[string]any.
// Unsupported types return an error; extending is straightforward when Phase 6 surfaces a real need.
//
// DSL-09: This is what powers `ctx.req.repo_name`-style dot access — req is converted to a Struct,
// and Starlark resolves .repo_name as an attribute lookup.
func ToStarlarkStruct(m map[string]any) (*starlarkstruct.Struct, error) {
    keys := make([]string, 0, len(m))
    for k := range m {
        keys = append(keys, k)
    }
    sort.Strings(keys) // ⚠ MANDATORY — Pitfall #3 / DSL-09

    sd := starlark.StringDict{}
    for _, k := range keys {
        v, err := toStarlarkValue(m[k])
        if err != nil {
            return nil, fmt.Errorf("ToStarlarkStruct: key %q: %w", k, err)
        }
        sd[k] = v
    }
    return starlarkstruct.FromStringDict(starlarkstruct.Default, sd), nil
}

func toStarlarkValue(v any) (starlark.Value, error) {
    switch x := v.(type) {
    case nil:
        return starlark.None, nil
    case string:
        return starlark.String(x), nil
    case int:
        return starlark.MakeInt(x), nil
    case int64:
        return starlark.MakeInt64(x), nil
    case float64:
        return starlark.Float(x), nil
    case bool:
        return starlark.Bool(x), nil
    case map[string]any:
        // Recurse into nested struct
        return ToStarlarkStruct(x)
    case []any:
        // Convert to *starlark.List (frozen — workflow state is read-only inside lambdas)
        elems := make([]starlark.Value, 0, len(x))
        for i, e := range x {
            sv, err := toStarlarkValue(e)
            if err != nil {
                return nil, fmt.Errorf("[%d]: %w", i, err)
            }
            elems = append(elems, sv)
        }
        lst := starlark.NewList(elems)
        lst.Freeze()
        return lst, nil
    default:
        return nil, fmt.Errorf("unsupported type %T", v)
    }
}

// pkg/bridge/lambda_call.go
package bridge

import (
    "context"
    "log/slog"

    "go.starlark.net/starlark"

    "github.com/mikelalcon/skytime/pkg/dag"
)

// CallOptions configures one CallLambda invocation. Defaults are tuned for D-22.
type CallOptions struct {
    MaxExecutionSteps uint64                                    // D-22: default 10_000_000 if zero
    PrintSink         func(ctx context.Context, msg string)     // D-21: default routes to slog.Default().Info
    Logger            *slog.Logger                              // D-06: optional, defaults to slog.Default()
    // Cancel is a channel signaling the caller wants the lambda interrupted.
    // Phase 3 wires this from workflow.Context.Done() via a watchdog goroutine.
    // Phase 1: nil-safe — when nil, no cancellation watchdog runs.
    Cancel <-chan struct{}
}

// DefaultMaxExecutionSteps is the D-22 default.
const DefaultMaxExecutionSteps uint64 = 10_000_000

// CallLambda evaluates a captured lambda with the given state.
// Always uses a FRESH *starlark.Thread (Pitfall #1) — never reuses across calls.
// Sets MaxExecutionSteps (D-22) and routes Print (D-21).
//
// Phase 1: this is testable standalone with hand-built CapturedLambda + state map.
// Phase 3: the interpreter wraps this with a watchdog goroutine that calls thread.Cancel
//          on workflow.Context.Done() and the PrintSink routes to workflow.GetLogger.Info.
func CallLambda(ctx context.Context, captured *dag.CapturedLambda, state map[string]any, opts CallOptions) (starlark.Value, error) {
    if opts.MaxExecutionSteps == 0 {
        opts.MaxExecutionSteps = DefaultMaxExecutionSteps
    }
    if opts.PrintSink == nil {
        logger := opts.Logger
        if logger == nil {
            logger = slog.Default()
        }
        opts.PrintSink = func(ctx context.Context, msg string) {
            logger.InfoContext(ctx, "starlark print", "msg", msg)
        }
    }

    // Convert state to *starlarkstruct.Struct (DSL-09)
    stateStruct, err := ToStarlarkStruct(state)
    if err != nil {
        return nil, err
    }

    // FRESH thread per call — Pitfall #1
    thread := &starlark.Thread{
        Name: "skytime-lambda:" + captured.ID,
        Print: func(_ *starlark.Thread, msg string) {
            opts.PrintSink(ctx, msg)
        },
    }
    thread.SetMaxExecutionSteps(opts.MaxExecutionSteps)

    // Watchdog: if Cancel fires, call thread.Cancel from a goroutine.
    // Phase 1 keeps it Go-native; Phase 3 inside workflow code uses workflow.Go.
    if opts.Cancel != nil {
        done := make(chan struct{})
        defer close(done)
        go func() {
            select {
            case <-opts.Cancel:
                thread.Cancel("cancelled by caller")
            case <-done:
            }
        }()
    }

    // Call the lambda with the state struct as the single positional arg.
    return starlark.Call(thread, captured.Fn, starlark.Tuple{stateStruct}, nil)
}

package bridge

import (

// MustFreeze freezes the given Starlark value and verifies that subsequent
// mutation attempts return errors containing "frozen". Test-only helper.
// Construct mutable Starlark values via starlark.NewDict / starlark.NewList,
// pass them through your code-under-test, then call MustFreeze to assert
// freeze semantics.
func MustFreeze(t *testing.T, v starlark.Value) {
    t.Helper()
    v.Freeze()

    switch x := v.(type) {
    case *starlark.Dict:
        err := x.SetKey(starlark.String("__freeze_test__"), starlark.None)
        if err == nil {
            t.Fatalf("dict was not frozen: SetKey succeeded")
        }
    case *starlark.List:
        err := x.Append(starlark.None)
        if err == nil {
            t.Fatalf("list was not frozen: Append succeeded")
        }
    }
}

    import (
        "fmt"

    // FromStarlarkValue is the inverse of toStarlarkValue. Phase 1 supports the basic types;
    // Phase 3's interpreter will extend as it consumes lambda outputs.
    func FromStarlarkValue(v starlark.Value) (any, error) {
        switch x := v.(type) {
        case starlark.NoneType:
            return nil, nil
        case starlark.String:
            return string(x), nil
        case starlark.Int:
            i, ok := x.Int64()
            if !ok {
                return nil, fmt.Errorf("int overflow")
            }
            return i, nil
        case starlark.Float:
            return float64(x), nil
        case starlark.Bool:
            return bool(x), nil
        case *starlark.List:
            out := make([]any, 0, x.Len())
            iter := x.Iterate()
            defer iter.Done()
            var elem starlark.Value
            for iter.Next(&elem) {
                g, err := FromStarlarkValue(elem)
                if err != nil {
                    return nil, err
                }
                out = append(out, g)
            }
            return out, nil
        case *starlark.Dict:
            out := make(map[string]any, x.Len())
            for _, item := range x.Items() {
                ks, ok := item[0].(starlark.String)
                if !ok {
                    return nil, fmt.Errorf("dict key must be string, got %s", item[0].Type())
                }
                gv, err := FromStarlarkValue(item[1])
                if err != nil {
                    return nil, err
                }
                out[string(ks)] = gv
            }
            return out, nil
        case *starlarkstruct.Struct:
            out := make(map[string]any, len(x.AttrNames()))
            for _, name := range x.AttrNames() {
                attr, err := x.Attr(name)
                if err != nil {
                    return nil, fmt.Errorf("struct.Attr(%q): %w", name, err)
                }
                gv, err := FromStarlarkValue(attr)
                if err != nil {
                    return nil, err
                }
                out[name] = gv
            }
            return out, nil
        default:
            return nil, fmt.Errorf("FromStarlarkValue: unsupported type %s", v.Type())
        }
    }
