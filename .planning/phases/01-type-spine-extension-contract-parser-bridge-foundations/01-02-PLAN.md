```go
// pkg/dag/node.go
package dag

import "go.starlark.net/syntax"

// Node is the sealed interface every DAG node implements.
// Phase 3's interpreter does `switch n := node.(type) { case *Flow: ...; case *Step: ... }`.
type Node interface {
    Kind() string                    // discriminator string ("Flow", "Step", "IfCond", ...)
    Position() syntax.Position       // .star call-site position for D-04 errors
    nodeMarker()                     // sealed — only types in pkg/dag can implement
}
```

```go
// pkg/dag/action.go — ActionRef is BOTH a dag node element AND a starlark.Value
// (Starlark sees ActionRef when an extension factory like `gh.create_issue(...)` returns it,
//  then the value is passed to step(action=...) and unwrapped on the Go side.)

import "go.starlark.net/starlark"

type ActionRef struct {
    Pos          syntax.Position
    Kind_        string                  // e.g. "github.create_issue" — D-08 fingerprint
    Kwargs       *starlark.Dict          // mutable-by-default; Freeze() must cascade
    CredentialID string
    frozen       bool
}

// Compile-time assertion that ActionRef satisfies starlark.Value.
var _ starlark.Value = (*ActionRef)(nil)

func (a *ActionRef) String() string         { return fmt.Sprintf("ActionRef(%s)", a.Kind_) }
func (a *ActionRef) Type() string           { return "ActionRef" }
func (a *ActionRef) Truth() starlark.Bool   { return starlark.True }
func (a *ActionRef) Hash() (uint32, error)  { return 0, fmt.Errorf("ActionRef is not hashable") }

// Freeze MUST cascade — Pitfall #6 prevention.
func (a *ActionRef) Freeze() {
    if a.frozen { return }
    a.frozen = true
    if a.Kwargs != nil {
        a.Kwargs.Freeze()
    }
}

// Convenience accessor (avoids exporting an unfortunate field name "Kind").
func (a *ActionRef) ActionKind() string { return a.Kind_ }
```

```go
    "time"

    "go.starlark.net/starlark"

type RetryPolicy struct {
    InitialInterval    time.Duration
    BackoffCoefficient float64
    MaxAttempts        int
    NonRetryableErrors []string
}

// Implements starlark.Unpacker so the step() builtin can do:
//   var retry *dag.RetryPolicy
//   starlark.UnpackArgs("step", ..., "retry?", &retry)
// and have RetryPolicy decode from a Starlark dict literal.
var _ starlark.Unpacker = (*RetryPolicy)(nil)

func (r *RetryPolicy) Unpack(v starlark.Value) error {
    d, ok := v.(*starlark.Dict)
    if !ok {
        return fmt.Errorf("retry must be a dict, got %s", v.Type())
    }
    // Iterate dict keys, parse known fields. Implementation is straightforward — see test
    // for expected behavior. Unknown keys → error (helps catch typos like "max_attempt" vs "max_attempts").
    return r.unpackDict(d)
}

type Timeout struct {
    StartToClose    time.Duration
    ScheduleToStart time.Duration
}

var _ starlark.Unpacker = (*Timeout)(nil)
func (t *Timeout) Unpack(v starlark.Value) error { /* same pattern as RetryPolicy */ }
```

    ```go
    package dag

    import (
        "fmt"
        "time"

        "go.starlark.net/starlark"
    )

    type RetryPolicy struct {
        InitialInterval    time.Duration
        BackoffCoefficient float64
        MaxAttempts        int
        NonRetryableErrors []string
    }

    var _ starlark.Unpacker = (*RetryPolicy)(nil)

    // Unpack decodes a Starlark dict like:
    //   {"initial_interval": "1s", "backoff_coefficient": 2.0, "max_attempts": 5, "non_retryable_errors": ["FOO"]}
    // into the receiver. Durations are parsed from Starlark strings via time.ParseDuration.
    func (r *RetryPolicy) Unpack(v starlark.Value) error {
        d, ok := v.(*starlark.Dict)
        if !ok {
            return fmt.Errorf("retry must be a dict, got %s", v.Type())
        }
        for _, item := range d.Items() {
            keyStr, ok := item[0].(starlark.String)
            if !ok {
                return fmt.Errorf("retry dict key must be string, got %s", item[0].Type())
            }
            key := string(keyStr)
            switch key {
            case "initial_interval":
                s, ok := item[1].(starlark.String)
                if !ok {
                    return fmt.Errorf("retry.initial_interval must be string (e.g. \"1s\"), got %s", item[1].Type())
                }
                dur, err := time.ParseDuration(string(s))
                if err != nil {
                    return fmt.Errorf("retry.initial_interval: %w", err)
                }
                r.InitialInterval = dur
            case "backoff_coefficient":
                f, ok := item[1].(starlark.Float)
                if !ok {
                    // also accept Int for "2" being equivalent to 2.0
                    if i, iok := item[1].(starlark.Int); iok {
                        if i64, ok := i.Int64(); ok {
                            r.BackoffCoefficient = float64(i64)
                            continue
                        }
                    }
                    return fmt.Errorf("retry.backoff_coefficient must be number, got %s", item[1].Type())
                }
                r.BackoffCoefficient = float64(f)
            case "max_attempts":
                i, ok := item[1].(starlark.Int)
                if !ok {
                    return fmt.Errorf("retry.max_attempts must be int, got %s", item[1].Type())
                }
                v, ok := i.Int64()
                if !ok {
                    return fmt.Errorf("retry.max_attempts overflow")
                }
                r.MaxAttempts = int(v)
            case "non_retryable_errors":
                lst, ok := item[1].(*starlark.List)
                if !ok {
                    return fmt.Errorf("retry.non_retryable_errors must be list, got %s", item[1].Type())
                }
                iter := lst.Iterate()
                defer iter.Done()
                var x starlark.Value
                for iter.Next(&x) {
                    s, ok := x.(starlark.String)
                    if !ok {
                        return fmt.Errorf("retry.non_retryable_errors entries must be string, got %s", x.Type())
                    }

        if !ok {
            return fmt.Errorf("timeout must be a dict, got %s", v.Type())
        }
        for _, item := range d.Items() {
            keyStr, ok := item[0].(starlark.String)
            if !ok {
                return fmt.Errorf("timeout dict key must be string, got %s", item[0].Type())
            }
            key := string(keyStr)
            valStr, ok := item[1].(starlark.String)
            if !ok {
                return fmt.Errorf("timeout.%s must be string (e.g. \"30s\"), got %s", key, item[1].Type())
            }
            dur, err := time.ParseDuration(string(valStr))
            if err != nil {
                return fmt.Errorf("timeout.%s: %w", key, err)
            }
            switch key {
            case "start_to_close":
                t.StartToClose = dur
            case "schedule_to_start":
                t.ScheduleToStart = dur
            default:
                return fmt.Errorf("timeout: unknown key %q (allowed: start_to_close, schedule_to_start)", key)
            }
        }
        return nil
    }
    ```
