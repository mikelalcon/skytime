// pkg/parser/options.go — functional options
package parser

import "github.com/mikelalcon/skytime/pkg/extension"

type Option func(*Parser)

func WithRoot(root string) Option {
    return func(p *Parser) { p.root = root }
}

func WithExtensions(exts ...extension.Extension) Option {
    return func(p *Parser) {
        for _, e := range exts {
            // Errors are deferred to NewParser's call site —
            // alternative: NewParser returns error.
            // Decision: NewParser returns error so registration failures are explicit.
            _ = p.registry.Register(e)
        }
    }
}

func WithMaxExecutionSteps(n uint64) Option {
    return func(p *Parser) { p.maxExecSteps = n }
}

// pkg/parser/globals.go
package parser

import (
    "go.starlark.net/starlark"

    "github.com/mikelalcon/skytime/pkg/extension"
)

// newParseTimeGlobals builds the rich parse-time predeclared environment:
// the six core DSL primitives + extension factories (per registry).
//
// PARSE-03: these globals are STRICTLY SUPERSET of bridge.LambdaTimeGlobals().
// Lambdas evaluated at execute time use a separate, more restricted dict.
func newParseTimeGlobals(p *Parser) starlark.StringDict {
    g := starlark.StringDict{
        "flow":              starlark.NewBuiltin("flow", p.builtinFlow),
        "step":              starlark.NewBuiltin("step", p.builtinStep),
        "if_cond":           starlark.NewBuiltin("if_cond", p.builtinIfCond),
        "script":            starlark.NewBuiltin("script", p.builtinScript),
        "for_each_parallel": starlark.NewBuiltin("for_each_parallel", p.builtinForEachParallel),
        "call_flow":         starlark.NewBuiltin("call_flow", p.builtinCallFlow),
    }
    // Inject extension factory values (e.g., "github" → github.endpoint factory).
    // Plan 03's Extension.Initialize is called per parse; the result is the value visible to .star.
    // Actually: plan 03 says Initialize is called when factory is invoked; here we register the extension
    // as a callable identifier. The Extension implementation is responsible for behaving as a Starlark value.
    // For now: each registered extension provides a starlark.Value (typically a *starlarkstruct.Module
    // or a *starlark.Builtin with the extension's name) accessible by Name() in parseTimeGlobals.
    // We assume Extension.Initialize is called lazily on first invocation. ALTERNATIVE: call Initialize
    // here and store the returned value. Decision: call Initialize here once per parser to keep things simple.
    for name, ext := range p.registry.All() {
        // Use a closure to capture ext; Initialize takes thread+kwargs but for parse-time-globals registration
        // we wrap it as a "factory builtin" that calls Initialize on its first invocation.
        g[name] = starlark.NewBuiltin(name, func(thread *starlark.Thread, fn *starlark.Builtin, args starlark.Tuple, kwargs []starlark.Tuple) (starlark.Value, error) {
            return ext.Initialize(thread, kwargs)
        })
    }
    return g
}

package parser

import (
    "fmt"

    "go.starlark.net/starlark"

    "github.com/mikelalcon/skytime/pkg/dag"
)

// builtinFlow implements: flow(name=..., inputs=..., steps=[...]) -> *dag.Flow
// The returned *dag.Flow is also captured into p.flows[name] (D-15: parser-session flow map).
func (p *Parser) builtinFlow(thread *starlark.Thread, fn *starlark.Builtin, args starlark.Tuple, kwargs []starlark.Tuple) (starlark.Value, error) {
    var (
        name   string
        inputs *starlark.Dict
        steps  *starlark.List
    )
    if err := starlark.UnpackArgs("flow", args, kwargs,
        "name", &name,
        "inputs?", &inputs,
        "steps", &steps,
    ); err != nil {
        return nil, p.wrapBuiltinError("flow", thread, err)
    }
    pos := callerPosition(thread)
    f := &dag.Flow{
        Pos: pos,
        Name: name,
        Inputs: convertInputsDict(inputs),
        Body: convertSteps(steps),
    }
    // D-15: track all flows in the session
    if existing, dup := p.flows[name]; dup {
        return nil, &dag.ParseError{
            Pos: pos,
            Msg: fmt.Sprintf("duplicate flow name %q (also defined at %s)", name, existing.Pos),
        }
    }
    p.flows[name] = f
    return f, nil
}

// CRITICAL: use thread.CallFrame(1).Pos, NOT fn.Position() (which would be the def site).
func callerPosition(thread *starlark.Thread) syntax.Position {
    if thread.CallStackDepth() < 2 {
        return syntax.Position{}

    return &dag.ParseError{Pos: pos, Msg: fmt.Sprintf("%s: %v", opName, err), Wrapped: err}

package parser

import (
    "fmt"
    "os"
    "path/filepath"
    "strings"

    "go.starlark.net/starlark"
    "go.starlark.net/syntax"

    "github.com/mikelalcon/skytime/pkg/dag"
)

func (p *Parser) makeLoad() func(*starlark.Thread, string) (starlark.StringDict, error) {
    return func(thread *starlark.Thread, module string) (starlark.StringDict, error) {
        callerFile := thread.CallFrame(1).Pos.Filename()
        resolved, err := p.resolveLoadPath(callerFile, module)
        if err != nil {
            return nil, err
        }
        if cached, ok := p.loadCache[resolved]; ok {
            return cached.globals, cached.err
        }
        src, err := os.ReadFile(resolved)
        if err != nil {
            return nil, &dag.ParseError{
                Pos: thread.CallFrame(1).Pos,
                Msg: fmt.Sprintf("load %q: %v", module, err),
            }
        }
        p.fileBytes[resolved] = src
        // FRESH thread for each loaded module — Pitfall #1
        loadThread := &starlark.Thread{
            Name: "load:" + resolved,
            Load: p.makeLoad(),
            Print: thread.Print,
        }
        loadThread.SetMaxExecutionSteps(p.maxExecSteps)
        opts := defaultFileOptions()
        globals, err := starlark.ExecFileOptions(opts, loadThread, resolved, src, p.parseTimeGlobals)
        if err != nil {
            err = wrapStarlarkError(err)
        }
        p.loadCache[resolved] = loadCacheEntry{globals: globals, err: err}
        return globals, err
    }
}

// pkg/parser/lambda_capture.go — D-18 stable IDs + D-19 free-var validation
package parser

import (
    "fmt"

    "go.starlark.net/starlark"
    "go.starlark.net/syntax"

    "github.com/mikelalcon/skytime/pkg/dag"
)

// captureLambda extracts a *starlark.Function, computes its D-18 stable ID,
// validates its free vars per D-19 (module-level + frozen only), and registers it
// in the parser session's lambda table.
func (p *Parser) captureLambda(thread *starlark.Thread, kwargName string, val starlark.Value) (*dag.CapturedLambda, error) {
    fn, ok := val.(*starlark.Function)
    if !ok {
        return nil, &dag.ParseError{
            Pos: callerPosition(thread),
            Msg: fmt.Sprintf("kwarg %q must be a lambda or function, got %s", kwargName, val.Type()),
        }
    }
    pos := fn.Position()
    fileBytes := p.fileBytes[pos.Filename()]
    if fileBytes == nil {
        // Defensive — should always be present after exec
        return nil, &dag.ParseError{
            Pos: pos,
            Msg: fmt.Sprintf("internal: file bytes for %q not cached", pos.Filename()),
        }
    }
    id := dag.ComputeLambdaID(fileBytes, pos)

    freeVars, err := p.validateFreeVars(fn, pos.Filename())
    if err != nil {
        return nil, err
    }
    captured := &dag.CapturedLambda{ID: id, Fn: fn, Pos: pos, FreeVars: freeVars}
    p.lambdas[id] = captured
    return captured, nil
}

// validateFreeVars rejects lambdas closing over non-module-level (mutable) variables.
func (p *Parser) validateFreeVars(fn *starlark.Function, ownerFile string) (starlark.StringDict, error) {
    out := starlark.StringDict{}
    for i := 0; i < fn.NumFreeVars(); i++ {
        binding, value := fn.FreeVar(i)
        if !isModuleLevelBinding(ownerFile, binding) {
            return nil, &dag.ParseError{
                Pos: binding.Pos,
                Msg: fmt.Sprintf("lambda captures non-module-level variable %q (free vars must reference module-level constants/functions only)", binding.Name),
            }
        }
        out[binding.Name] = value
    }
    out.Freeze()
    return out, nil
}

// isModuleLevelBinding: true when binding.Pos.Col == 1 in the same file (top-level scope).
func isModuleLevelBinding(filename string, binding starlark.Binding) bool {
    return binding.Pos.Filename() == filename && binding.Pos.Col == 1
}

// pkg/parser/errors.go — wrap go.starlark.net errors into *dag.ParseError
package parser

import (
    "errors"
    "go.starlark.net/starlark"
    "go.starlark.net/syntax"
    "github.com/mikelalcon/skytime/pkg/dag"
)

// wrapStarlarkError converts go.starlark.net errors (*starlark.EvalError, syntax.Error)
// into *dag.ParseError preserving position info. Per PARSE-05.
func wrapStarlarkError(err error) error {
    if err == nil { return nil }
    var pe *dag.ParseError
    if errors.As(err, &pe) { return err } // already wrapped
    var evalErr *starlark.EvalError
    if errors.As(err, &evalErr) {
        var pos syntax.Position
        if len(evalErr.CallStack) > 0 {
            pos = evalErr.CallStack[0].Pos
        }
        return &dag.ParseError{Pos: pos, Msg: evalErr.Msg + "\n" + evalErr.Backtrace(), Wrapped: err}
    }
    var syntaxErr syntax.Error
    if errors.As(err, &syntaxErr) {
        return &dag.ParseError{Pos: syntaxErr.Pos, Msg: syntaxErr.Msg, Wrapped: err}
    }
    return &dag.ParseError{Msg: err.Error(), Wrapped: err}
}

    func (p *Parser) resolveLoadPath(callerFile, module string) (string, error) {
        // Determine root
        root := p.root
        if root == "" {
            root = findGitRoot(filepath.Dir(callerFile))
        }
        var candidate string
        switch {
        case strings.HasPrefix(module, "/"):
            // Absolute path (D-13): from configured root
            if root == "" {
                return "", &dag.ParseError{Msg: fmt.Sprintf("absolute load path %q used but no root configured (set WithRoot or place files under a .git ancestor)", module)}
            }
            candidate = filepath.Join(root, module)
        case strings.HasPrefix(module, "./") || strings.HasPrefix(module, "../"):
            // Relative path (D-13): sibling of caller file
            candidate = filepath.Join(filepath.Dir(callerFile), module)
        default:
            // Bare name → treat as relative
            candidate = filepath.Join(filepath.Dir(callerFile), module)
        }
        abs, err := filepath.Abs(candidate)
        if err != nil {
            return "", &dag.ParseError{Msg: fmt.Sprintf("resolve load %q: %v", module, err)}
        }
        // D-17 sandbox: must stay within root (if root is set)
        if root != "" {
            rootAbs, err := filepath.Abs(root)
            if err != nil {
                return "", &dag.ParseError{Msg: fmt.Sprintf("resolve root %q: %v", root, err)}
            }
            rel, err := filepath.Rel(rootAbs, abs)
            if err != nil || strings.HasPrefix(rel, "..") {
                return "", &dag.ParseError{Msg: fmt.Sprintf("load %q: path escapes parser root %q", module, rootAbs)}
            }
        }
        return abs, nil
    }

            }

                return ""
            }

    import (

    // validateFreeVars walks every free variable of a captured lambda and rejects
    // any that doesn't resolve to a module-level (top-of-file) binding. Per D-19:
    // "lambdas may close over module-level constants/functions only".
    func (p *Parser) validateFreeVars(fn *starlark.Function, ownerFile string) (starlark.StringDict, error) {
        out := starlark.StringDict{}
        for i := 0; i < fn.NumFreeVars(); i++ {
            binding, value := fn.FreeVar(i)
            if !isModuleLevelBinding(ownerFile, binding) {
                return nil, &dag.ParseError{
                    Pos: binding.Pos,
                    Msg: fmt.Sprintf("lambda captures non-module-level variable %q (free vars must reference module-level constants/functions only)", binding.Name),
                }
            }
            out[binding.Name] = value
        }
        out.Freeze()
        return out, nil
    }

    // isModuleLevelBinding returns true when binding originates at the top of the file (column 1).
    // A def-declared helper at indent zero is module-level; a let-binding inside a def body is not.
    func isModuleLevelBinding(filename string, binding *starlark.Binding) bool {
        if binding == nil {
            return false
        }
        // Note: Starlark's Binding.Pos.Col == 1 means the binding statement starts at column 1
        // (top-of-file scope). A def-declared helper IS at column 1 even though its body indents.
        return binding.Pos.Filename() == filename && binding.Pos.Col == 1
    }
