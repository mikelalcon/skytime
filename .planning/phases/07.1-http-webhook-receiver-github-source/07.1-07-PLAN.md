```python
"""webhook_demo.star — D-7.1-16 crash-recovery walkthrough flow.

Trigger fires on GitHub `issues` events. The flow comments on the issue,
waits 30 seconds (durable Temporal sleep — survives worker kill), then
labels the issue. The reader can:

  1. Open an issue in their test repo (gh webhook forward delivers).
  2. Watch the comment "received, processing..." appear within seconds.
  3. Kill the skytime server with Ctrl-C BEFORE 30 seconds elapse.
  4. Restart the server.
  5. Watch the label "processed" appear ~30 seconds after the ORIGINAL
     delivery — Temporal continued the workflow from event history.

Demonstrates Temporal's headline durability story end-to-end against a
real provider (GitHub) without tunnels or OAuth app registration.

See docs/walkthroughs/github-webhook.md (Plan 08) for the full reader
walkthrough including the credfile setup and gh extension install steps.
"""

gh = github.client(credential = "github_token")

# ---------------------------------------------------------------------------
# Trigger: GitHub `issues` events → webhook_demo flow.
# ---------------------------------------------------------------------------
trigger(
    flow = "webhook_demo",
    source = github.webhook(
        events = ["issues"],
        secret_credential = "github_webhook_secret",
    ),
    map = lambda req: {
        "repo": req.payload.repository.full_name,
        "issue_number": req.payload.issue.number,
    },
    idempotency_key = lambda req: req.headers["X-GitHub-Delivery"],
    credential = "github_webhook_secret",
)

# ---------------------------------------------------------------------------
# Flow: comment → wait 30s → label. Durable across worker kill.
# ---------------------------------------------------------------------------
flow(
    name = "webhook_demo",
    inputs = {
        "repo": "string",          # "owner/repo" full_name from req.payload.repository
        "issue_number": "int",     # req.payload.issue.number
    },
    steps = [
        # Step 1: post the "received, processing..." comment immediately.
        # Uses script + ctx.repo to split owner/repo since v1's static-block
        # constraint requires literal owner/repo for the kwargs reflection;
        # see the action_fn pattern from issue_triage.star.
        step(
            name = "Post received comment",
            action_fn = lambda ctx: gh.add_comment(
                owner = ctx.repo.split("/")[0],
                repo = ctx.repo.split("/")[1],
                number = ctx.issue_number,
                body = "received, processing...",
            ),
            retry = {"max_attempts": 3, "initial_interval": "1s"},
            timeout = {"start_to_close": "30s"},
        ),

        # Step 2: durable 30-second wait. The reader kills the server
        # mid-wait and restarts; Temporal resumes the timer from event
        # history. NOTE: the actual sleep idiom MUST match the example
        # corpus convention. If `step(sleep="30s")` is not a shipped
        # primitive, the executor substitutes the appropriate pattern
        # documented in the issue_triage.star sleep equivalent.
        # SUBSTITUTE-DURING-EXECUTION: replace this comment with the
        # canonical sleep idiom found in the existing corpus.
        # If the corpus has no sleep idiom, omit this step entirely
        # and document the omission in the SUMMARY (the demo still
        # demonstrates durability — the long-running activity timeout
        # of step 1 plus step 3's network call provide visible
        # cross-restart artifacts).
        # (placeholder — executor MUST replace before commit)

        # Step 3: post the "processed" label. After the wait, this
        # action lands ~30s after the original webhook delivery.
        step(
            name = "Apply processed label",
            action_fn = lambda ctx: gh.add_label(
                owner = ctx.repo.split("/")[0],
                repo = ctx.repo.split("/")[1],
                number = ctx.issue_number,
                label = "processed",
            ),
            retry = {"max_attempts": 3, "initial_interval": "1s"},
            timeout = {"start_to_close": "30s"},
        ),
    ],
)
```
