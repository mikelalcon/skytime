```go
// Deps is the dependency bundle Mount needs at request time. Populated
// by pkg/cli/server.go (Plan 06) before calling Mount.
type Deps struct {
	// Client is the Temporal client used for ExecuteWorkflow. Required.
	Client client.Client

	// CredentialHandler resolves credential IDs JIT inside the request
	// handler. Same handler the activity layer uses (Phase 2). Required.
	// May be a no-op handler if all configured triggers have empty
	// secretCredID — the handler short-circuits on empty ID.
	CredentialHandler extension.CredentialHandler

	// TaskQueue is the Temporal task queue ExecuteWorkflow targets.
	// Must match the Worker's TaskQueue. Required.
	TaskQueue string

	// Logger is the per-request log sink. Same logger as the server
	// startup banner (charm-log default; --json-log routes to JSON).
	// Required.
	Logger *slog.Logger
}

// validate panics on missing required Deps fields. Boot-time programmer
// error; not a runtime concern.
func (d Deps) validate() {
	if d.Client == nil {
		panic("receiver.Deps: Client is required")
	}
	if d.CredentialHandler == nil {
		panic("receiver.Deps: CredentialHandler is required (use a no-op handler for unsigned mounts)")
	}
	if d.TaskQueue == "" {
		panic("receiver.Deps: TaskQueue is required")
	}
	if d.Logger == nil {
		panic("receiver.Deps: Logger is required")
	}
}

// mountKey groups triggers by (source kind, path, method) per D-7.1-06.
// Two triggers with the same key fan out from one handler.
type mountKey struct {
	kind   string
	path   string
	method string
}

// Mount walks worker.Triggers().All(), groups HTTP-shaped triggers by
// mountKey, and registers ONE handler per group on the supplied mux.
//
// Cron / non-HTTP sources are silently skipped (they don't implement
// receiver.HTTPMounter; Phase 7.2 owns their dispatch path).
//
// Per D-7.1-06 fan-out: when N triggers subscribe to the same mount
// key, they ALL fire on every matching delivery. Per-trigger event
// filtering (github.webhook ShouldDispatch) further narrows the
// dispatched set inside the handler.
func Mount(mux *http.ServeMux, w *worker.Worker, deps Deps) {
	deps.validate()

	groups := map[mountKey][]*dag.Trigger{}
	for _, t := range w.Triggers().All() {
		mounter, ok := t.Source.(HTTPMounter)
		if !ok {
			continue // cron sources, etc.
		}
		path, method := mounter.HTTPMount()
		key := mountKey{kind: t.Source.Kind(), path: path, method: method}
		groups[key] = append(groups[key], t)
	}

	for key, trigs := range groups {
		// Stable sort within group: (FlowName, Pos.String). Plan 04's
		// TriggerRegistry.Freeze already sorts the registry, but
		// re-sorting per group makes the within-group order
		// independent of how the iteration above happens.
		sort.SliceStable(trigs, func(i, j int) bool {
			if trigs[i].FlowName != trigs[j].FlowName {
				return trigs[i].FlowName < trigs[j].FlowName
			}
			return trigs[i].Pos.String() < trigs[j].Pos.String()
		})
		mux.HandleFunc(key.path, methodGate(key.method, makeHandler(key, trigs, deps)))
	}
}

// methodGate returns 405 Method Not Allowed for non-matching methods.
// Stdlib http.ServeMux does NOT method-dispatch; we layer it here so
// /x with mounted POST handler still returns 405 on GET (instead of
// the default — which would be the POST handler firing).
func methodGate(want string, next http.HandlerFunc) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		if r.Method != want {
			http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
			return
		}
		next(w, r)
	}
}
```

```go
// makeHandler returns the per-mount http.HandlerFunc that processes one
// delivery and dispatches across the trigs fan-out slice.
//
// LOCKED pipeline (D-7.1-14, D-7.1-15):
//
//	1. defer requestRecord.emit (always emitted, even on panic)
//	2. body size cap via http.MaxBytesReader (D-7.1-12)
//	3. read raw body bytes via io.ReadAll (Pitfall 2 — HMAC vs raw bytes)
//	4. JIT credential resolve via deps.CredentialHandler.Resolve (per request, D-7.1)
//	5. signature validation via validateHMAC (Plan 01)
//	6. content-type check (415 on non-JSON)
//	7. json.Unmarshal body to map[string]any
//	8. construct req map (payload + headers) for lambda eval
//	9. fan-out: event filter → eval map+idempotency → composeWorkflowID
//	10. ExecuteWorkflow with REJECT_DUPLICATE + WorkflowExecutionErrorWhenAlreadyStarted=true
//	11. classify per-trigger result; pick worst status; write response
func makeHandler(key mountKey, trigs []*dag.Trigger, deps Deps) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		rec := &requestRecord{
			method:     r.Method,
			path:       r.URL.Path,
			start:      time.Now(),
			sourceKind: key.kind,
			errorClass: errorClassOK,
		}
		// Status default — overwritten before final emit. Keeps log
		// records consistent on early-return paths.
		rec.status = http.StatusOK

		// to net/http's default handler which writes 500 but skips our
		// deferred logger. Recover here so the log line still emits.
		defer func() {
			if pv := recover(); pv != nil {
				rec.status = http.StatusInternalServerError
				rec.errorClass = errorClassLambdaPanic
				writeInternalResponse(w, "lambda_panic")
			}
			rec.emit(deps.Logger)
		}()

		r.Body = http.MaxBytesReader(w, r.Body, maxBodyBytes)
		bodyBytes, err := io.ReadAll(r.Body)
		if err != nil {
			// http.MaxBytesReader has already written the response on
			// overflow. We just record the log fields.
			rec.status = http.StatusRequestEntityTooLarge
			rec.errorClass = errorClassBadRequest
			return
		}

		// 2. event header — only meaningful for github.webhook.
		eventName := ""
		if key.kind == skygh.GithubWebhookKind {
			eventName = r.Header.Get(skygh.GithubWebhookEventHeader)
			rec.event = eventName
		}

		// 3. signature validation. Determine the signing path based on
		// the source's kind + (algo, header, secretCredID). All
		// triggers in this group share the SAME mount-level config
		// (D-7.1-06 fan-out groups by kind+path+method) — so we read
		// signing config from the FIRST trigger in trigs. The kind
		// dictates which accessor methods to use.
		algo, sigHeader, secretCredID := readSigningConfig(key.kind, trigs[0])
		if secretCredID != "" {
			cred, err := deps.CredentialHandler.Resolve(r.Context(), secretCredID)
			if err != nil {
				rec.status = http.StatusInternalServerError
				rec.errorClass = errorClassDispatchFailed
				writeInternalResponse(w, "credential_resolve_failed")
				return
			}
			secretBytes, ok := credentialBytes(cred)
			if !ok {
				rec.status = http.StatusInternalServerError
				rec.errorClass = errorClassDispatchFailed
				writeInternalResponse(w, "credential_unsupported_kind")
				return
			}

			// It NEVER enters the log line, NEVER enters the response,
			// NEVER enters errors. Used ONLY for the HMAC computation.
			headerValue := r.Header.Get(sigHeader)

				rec.status = http.StatusUnauthorized
				rec.errorClass = errorClassSignatureMismatch
				writeUnauthorizedResponse(w)
				return
			}

		// D-7.1-05. 415 on non-JSON (Body decoder negotiation

		// for it, so this path is for users who misconfigure their

// readSigningConfig returns the (algo, header, secretCredID) tuple for
// the first trigger in a fan-out group. All triggers in a group share
// the same mount key, but http.webhook configs (algo, header) can
// differ across triggers binding the same path/method — in v1 we
// reject mixed configs at parse time (TODO: leave parse-time check
// as future tightening; for now assume the first trigger's config
// is the group's config).
func readSigningConfig(kind string, t *dag.Trigger) (algo, header, secretCredID string) {
	switch kind {
	case skygh.GithubWebhookKind:
		algo = skygh.GithubWebhookSignatureAlgo
		header = skygh.GithubWebhookSignatureHeader
		if g, ok := t.Source.(interface{ SecretCredID() string }); ok {
			secretCredID = g.SecretCredID()
		}
	default:
		// Assumes http.webhook (the only other HTTP-shaped source in 7.1).
		if h, ok := t.Source.(interface {
			SignatureAlgo() string
			SignatureHeader() string
			SecretCredID() string
		}); ok {
			algo = h.SignatureAlgo()
			header = h.SignatureHeader()
			secretCredID = h.SecretCredID()
		}
	}
	return
}

// credentialBytes returns the raw bytes for HMAC computation depending
// on the credential kind. Bearer.Token, Basic.Password, APIKey.Key all
// expose .Reveal() through their Secret field.
//
// CRITICAL: this is a Reveal() call site. Each one is a leak boundary.
// The returned bytes go ONLY to validateHMAC; they never leave this
// handler scope.
func credentialBytes(cred extension.Credential) ([]byte, bool) {
	switch c := cred.(type) {
	case *extension.BearerCredential:
		return []byte(c.Token.Reveal()), true
	case *extension.BasicCredential:
		return []byte(c.Password.Reveal()), true
	case *extension.APIKeyCredential:
		return []byte(c.Key.Reveal()), true
	default:
		return nil, false
	}
}

// flattenHeaders converts http.Header (multi-value) to a flat
// map[string]any keyed by canonical header name. Multi-value headers
// take their FIRST value (webhook providers send single-valued
// headers in practice).
func flattenHeaders(h http.Header) map[string]any {
	out := make(map[string]any, len(h))
	for k, vv := range h {
		if len(vv) > 0 {
			out[k] = vv[0]
		}
	}
	return out
}

// evalLambdaToMap evaluates a CapturedLambda against req as the lone
// state key, expecting a starlark dict result that converts to
// map[string]any.
func evalLambdaToMap(ctx context.Context, lam *dag.CapturedLambda, req map[string]any) (map[string]any, error) {
	state := map[string]any{"req": req}
	v, err := bridge.CallLambda(ctx, lam, state, bridge.CallOptions{
		Globals: bridge.TriggerTimeGlobals(),
	})
	if err != nil {
		return nil, err
	}
	d, ok := v.(*starlark.Dict)
	if !ok {
		return nil, fmt.Errorf("map lambda must return dict, got %s", v.Type())
	}
	return starlarkDictToMap(d)
}

// evalLambdaToString evaluates a CapturedLambda expecting a string result.
func evalLambdaToString(ctx context.Context, lam *dag.CapturedLambda, req map[string]any) (string, error) {
	state := map[string]any{"req": req}
	v, err := bridge.CallLambda(ctx, lam, state, bridge.CallOptions{
		Globals: bridge.TriggerTimeGlobals(),
	})
	if err != nil {
		return "", err
	}
	s, ok := starlark.AsString(v)
	if !ok {
		return "", fmt.Errorf("idempotency_key lambda must return string, got %s", v.Type())
	}
	return s, nil
}

// starlarkDictToMap is a minimal converter for the map-lambda return.
// More elaborate cases (nested dicts, lists) are handled lazily — for
// v1, workflow inputs are flat string-keyed maps with primitive values.
func starlarkDictToMap(d *starlark.Dict) (map[string]any, error) {
	out := make(map[string]any, d.Len())
	for _, item := range d.Items() {
		key, ok := starlark.AsString(item.Index(0))
		if !ok {
			return nil, fmt.Errorf("dict key must be string, got %s", item.Index(0).Type())
		}
		val, err := starlarkValueToGo(item.Index(1))
		if err != nil {
			return nil, fmt.Errorf("dict[%q]: %w", key, err)
		}
		out[key] = val
	}
	return out, nil
}

// starlarkValueToGo converts primitive starlark values to Go any.
// Sufficient for v1's flat workflow-input map.
func starlarkValueToGo(v starlark.Value) (any, error) {
	switch x := v.(type) {
	case starlark.String:
		return string(x), nil
	case starlark.Int:
		i64, _ := x.Int64()
		return i64, nil
	case starlark.Float:
		return float64(x), nil
	case starlark.Bool:
		return bool(x), nil
	case starlark.NoneType:
		return nil, nil
	default:
		// Fallback: stringify. Lambdas returning lists/dicts in v1
		// degrade to strings; future work expands this.
		return v.String(), nil
	}
}
```
