```go
//go:build !windows

package cli

import (
    "fmt"
    "io"
    "strings"
    "sync"
    "time"
)

// braille frames for the spinner — npm/cargo de-facto standard set.
var spinnerFrames = []string{"⠋", "⠙", "⠹", "⠸", "⠼", "⠴", "⠦", "⠧", "⠇", "⠏"}

// activeStep is one in-progress row in the live block's redraw region.
type activeStep struct {
    Idx       int64
    Total     int64
    Kind      string
    Label     string
    StartedAt time.Time
}

// liveRenderer owns all writes to its output writer. Slog Handle()
// invocations push events through eventsCh; the single goroutine
// running run() pops, applies, and redraws. The 100 ms ticker drives
// the spinner cadence (D4.1-18) and acts as a debouncer for rapid
// event bursts (Pitfall 7 from RESEARCH).
type liveRenderer struct {
    out    io.Writer
    events chan progressEvent
    closed chan struct{}
    once   sync.Once
    wg     sync.WaitGroup

    // State owned by the render goroutine ONLY.
    active     []*activeStep
    cap        int
    drawnLines int
    spinIdx    int
}

// progressEvent is the immutable value the slog handler ships to the
// renderer. Fully-owned per-event; the handler creates and forgets.
type progressEvent struct {
    Kind      string // "flow_start" | "step_dispatch" | "step_complete" | "branch" | "flow_complete" | "raw"
    FlowName  string
    StepCount int64
    Idx       int64
    Total     int64
    KindAttr  string // event["kind"] — "step", "if_cond", etc.
    Label     string
    Status    string
    Summary   string
    DurationMs int64
    OkCount   int64
    ErrCount  int64
    TotalMs   int64
    Branch    string
    Path      string
    Raw       string // For "raw" events that should pass through to wrapped handler — pre-rendered line
}

func newLiveRenderer(out io.Writer) *liveRenderer {
    r := &liveRenderer{
        out:    out,
        events: make(chan progressEvent, 64),
        closed: make(chan struct{}),
        cap:    10, // D4.1-19
    }
    // Hide cursor on activation; restored on Close().
    fmt.Fprint(out, "\x1b[?25l")
    r.wg.Add(1)
    go r.run()
    return r
}

func (r *liveRenderer) submit(ev progressEvent) {
    select {
    case r.events <- ev:
    case <-r.closed:
    }
}

// Close drains pending events, clears the redraw region, restores the
// cursor, and waits for the render goroutine to exit.
func (r *liveRenderer) Close() {
    r.once.Do(func() {
        close(r.events)
        r.wg.Wait()
        // Show cursor.
        fmt.Fprint(r.out, "\x1b[?25h")
    })
}

func (r *liveRenderer) run() {
    defer r.wg.Done()
    tick := time.NewTicker(100 * time.Millisecond) // D4.1-18
    defer tick.Stop()
    for {
        select {
        case ev, ok := <-r.events:
            if !ok {
                r.flushFinal()
                return
            }
            r.applyEvent(ev)
            r.redraw()
        case <-tick.C:
            r.spinIdx = (r.spinIdx + 1) % len(spinnerFrames)
            r.redraw()
        }
    }
}

// applyEvent updates the active list based on event kind. For
// flow_start and flow_complete, applies the printed-line semantics
// (header / footer accumulate above the redraw region).
func (r *liveRenderer) applyEvent(ev progressEvent) {
    switch ev.Kind {
    case "flow_start":
        // Clear current redraw region (none expected at flow start)
        // and emit the header as a STATIC line above the future
        // redraw region.
        r.clearRedrawRegion()
        fmt.Fprintf(r.out, "[skytime] flow %s  %d steps  starting\n", ev.FlowName, ev.StepCount)
    case "step_dispatch":
        r.active = append(r.active, &activeStep{
            Idx: ev.Idx, Total: ev.Total, Kind: ev.KindAttr, Label: ev.Label,
            StartedAt: time.Now(),
        })
    case "step_complete":
        // Find and remove the matching active row; emit the static
        // ✓/✗ line above the redraw region.
        r.clearRedrawRegion()
        for i, s := range r.active {
            if s.Idx == ev.Idx {
                r.active = append(r.active[:i], r.active[i+1:]...)
                break
            }
        }
        marker := "✓"
        if ev.Status == "err" {
            marker = "✗"
        }
        fmt.Fprintf(r.out, "[%d/%d] step  %s %dms  %s\n", ev.Idx, ev.Total, marker, ev.DurationMs, ev.Summary)
    case "branch":
        r.clearRedrawRegion()
        fmt.Fprintf(r.out, "     → %s\n", ev.Branch)
    case "flow_complete":
        r.clearRedrawRegion()
        if ev.ErrCount > 0 {
            fmt.Fprintf(r.out, "[skytime] flow failed  total %dms\n", ev.TotalMs)
        } else {
            fmt.Fprintf(r.out, "[skytime] flow complete  %d/%d steps  total %dms\n", ev.OkCount, ev.OkCount+ev.ErrCount, ev.TotalMs)
        }
    case "raw":
        r.clearRedrawRegion()
        fmt.Fprintln(r.out, ev.Raw)
    }
}

// clearRedrawRegion moves the cursor up r.drawnLines and erases each
// line. Resets r.drawnLines to 0.
func (r *liveRenderer) clearRedrawRegion() {
    for i := 0; i < r.drawnLines; i++ {
        fmt.Fprint(r.out, "\x1b[1A\x1b[2K")
    }
    r.drawnLines = 0
}

// redraw paints the current active list as the redraw region. Called
// on every tick AND after applyEvent. Idempotent: clearRedrawRegion +
// emit-rows.
func (r *liveRenderer) redraw() {
    r.clearRedrawRegion()
    n := len(r.active)
    if n == 0 {
        return
    }
    visible := n
    if visible > r.cap {
        visible = r.cap
    }
    fmt.Fprintf(r.out, "[skytime] in-progress  %d active\n", n)
    drawn := 1
    for i := 0; i < visible; i++ {
        s := r.active[i]
        elapsed := time.Since(s.StartedAt).Seconds()
        // Truncate label to keep line under typical 80-col width.
        label := s.Label
        if len(label) > 60 {
            label = label[:57] + "..."
        }
        fmt.Fprintf(r.out, "  [%d/%d] %s  %s  %s %.1fs\n",
            s.Idx, s.Total, padKind(s.Kind), label, spinnerFrames[r.spinIdx], elapsed)
        drawn++
    }
    if n > r.cap {
        fmt.Fprintf(r.out, "  ... and %d more\n", n-r.cap)
        drawn++
    }
    r.drawnLines = drawn
}

func (r *liveRenderer) flushFinal() {
    r.clearRedrawRegion()
}

// stripAnsi is a test helper exported via a build-tag-free file (see
// progress_testutil_test.go).
func stripAnsi(s string) string {
    var b strings.Builder
    for i := 0; i < len(s); {
        if s[i] == '\x1b' && i+1 < len(s) && s[i+1] == '[' {
            // Skip until alpha
            j := i + 2
            for j < len(s) && !((s[j] >= 'A' && s[j] <= 'Z') || (s[j] >= 'a' && s[j] <= 'z')) {
                j++
            }
            i = j + 1
            continue
        }
        b.WriteByte(s[i])
        i++
    }
    return b.String()
}
```
