And add a helper:
```go
// desugarActionRefKwargs walks ar.Kwargs and replaces every
// starlark.String value containing "${" with a *StarlarkLambda
// produced by desugarInterpolation. Acts in-place on the (still
// unfrozen) dict. ar.Pos is used as openPos for the desugarer; the
// scanner's per-${ position adjustment computes the precise location
// of each marker within the string value.
func (p *Parser) desugarActionRefKwargs(ar *dag.ActionRef) error {
    if ar == nil || ar.Kwargs == nil {
        return nil
    }
    for _, item := range ar.Kwargs.Items() {
        keyStr, _ := item[0].(starlark.String)
        valStr, isStr := item[1].(starlark.String)
        if !isStr {
            continue
        }
        if !strings.Contains(string(valStr), "${") {
            continue
        }
        captured, err := p.desugarInterpolation(string(valStr), ar.Pos)
        if err != nil {
            return err
        }
        if captured == nil {
            // Only $$ escapes encountered — replace string with the
            // de-escaped literal so the activity sees the unescaped form.
            // Implementation: re-run scanInterpolation, render the text
            // parts. For simplicity, we COULD leave the literal alone
            // since $$ outside ${...} is uncommon — but the spec says
            // "$$" must produce literal "$" at runtime, so the rewrite
            // is needed for correctness. Defer to a TestEscape fixture
            // for coverage.
            scanned, _ := scanInterpolation(string(valStr), ar.Pos)
            var rebuilt strings.Builder
            for _, part := range scanned.Parts {
                rebuilt.WriteString(part.Text)
            }
            if err := ar.Kwargs.SetKey(keyStr, starlark.String(rebuilt.String())); err != nil {
                return err
            }
            continue
        }
        if err := ar.Kwargs.SetKey(keyStr, dag.NewStarlarkLambda(captured)); err != nil {
            return err
        }
    }
    return nil
}
```
