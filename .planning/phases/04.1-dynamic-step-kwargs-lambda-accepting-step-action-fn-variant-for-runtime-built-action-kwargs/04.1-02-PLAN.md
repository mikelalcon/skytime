// buildLambdaSource turns a sequence of scanPart into the body of a
// `lambda ctx: ...` expression. The synthesized source ALWAYS wraps
// each expr in str(...) per D4.1-04 — the wrap is unconditional for
// safety (handles int, bool, None, float gracefully).
//
// For an empty parts list this returns `""` — the caller is expected
// to short-circuit before calling this function (no interpolation
// means no synthesized lambda).
func buildLambdaSource(parts []scanPart) string {
    var b strings.Builder
    b.WriteString("lambda ctx: ")
    if len(parts) == 0 {
        b.WriteString(`""`)
        return b.String()
    }
    for i, p := range parts {
        if i > 0 {
            b.WriteString(" + ")
        }
        switch p.Kind {
        case "text":
            // Re-quote as a Starlark string literal — must escape
            // backslashes, quotes, newlines.
            b.WriteString(starlarkQuote(p.Text))
        case "expr":
            b.WriteString("str(")
            b.WriteString(p.Expr)
            b.WriteString(")")
        }
    }
    return b.String()
}

// desugarInterpolation converts a Starlark string literal's content
// into a synthesized *CapturedLambda. Returns (nil, nil) when the
// input contains no ${...} markers (caller stores the literal string
// unchanged). Returns (nil, *dag.ParseError) on scanner errors or
// invalid inner expressions.
//
// openPos is the position of the OPENING ${ in the user's source —
// used for Pos (error attribution) AND lambda ID computation.
//
// Implementation steps (RESEARCH §Pattern 1):
//   1. Scan via scanInterpolation; short-circuit if !HasInterpolation.
//   2. For each expr part, validate via syntax.ParseExpr — surface
//      syntax errors with the user's position.
//   3. Build "result = " + buildLambdaSource(parts).
//   4. Cache the synthetic source under "<interp:<file>:<line>:<col>>"
//      in p.fileBytes so D4-02's findCtxAccesses can re-parse it.
//   5. Compile via starlark.ExecFileOptions on a sub-thread that uses
//      lambdaTimeGlobals (the locked 20-key D-20 subset).
//   6. Pluck globals["result"] as *starlark.Function.
//   7. Build *CapturedLambda with:
//        - ID = ComputeLambdaID(p.fileBytes[openPos.Filename()], openPos)
//        - Fn = the synthesized function
//        - Pos = openPos (user attribution)
//        - BodyPos = the synthetic position (Line=1, Col=10 — past "result = ")
//        - FreeVars = empty StringDict (synthesized lambdas have no free vars)
//   8. Register in p.lambdas[id].
//   9. Return the captured lambda.
func (p *Parser) desugarInterpolation(raw string, openPos syntax.Position) (*dag.CapturedLambda, error) {
    scanned, err := scanInterpolation(raw, openPos)
    if err != nil {
        return nil, err
    }
    if !scanned.HasInterpolation {
        return nil, nil
    }
    // Validate each expr part via syntax.ParseExpr — surfaces syntax
    // errors at the user's source position rather than the synthetic file.
    for _, part := range scanned.Parts {
        if part.Kind != "expr" {
            continue
        }
        if _, perr := syntax.ParseExpr(openPos.Filename(), part.Expr, 0); perr != nil {
            return nil, &dag.ParseError{
                Pos: part.Pos,
                Msg: fmt.Sprintf("interpolation expression %q: %v", part.Expr, perr),
            }
        }
    }
    src := buildLambdaSource(scanned.Parts)
    syntheticName := fmt.Sprintf("<interp:%s:%d:%d>", openPos.Filename(), openPos.Line, openPos.Col)
    fullSrc := "result = " + src
    p.fileBytes[syntheticName] = []byte(fullSrc)

    // Compile in a sub-thread with the lambda-time globals (D-20).
    // The sub-thread inherits NOTHING from the parser thread — clean room.
    subThread := &starlark.Thread{Name: "skytime-desugar:" + syntheticName}
    opts := defaultFileOptions()
    globals, err := starlark.ExecFileOptions(opts, subThread, syntheticName, fullSrc, lambdaTimeGlobalsForDesugar())
    if err != nil {
        return nil, &dag.ParseError{
            Pos: openPos,
            Msg: fmt.Sprintf("interpolation: failed to compile synthesized lambda: %v", err),
        }
    }
    fnVal, ok := globals["result"]
    if !ok {
        return nil, &dag.ParseError{Pos: openPos, Msg: "interpolation: internal — synthesized source produced no 'result'"}
    }
    fn, ok := fnVal.(*starlark.Function)
    if !ok {
        return nil, &dag.ParseError{Pos: openPos, Msg: fmt.Sprintf("interpolation: internal — synthesized 'result' is %s, expected function", fnVal.Type())}
    }

    return p.captureLambdaAtPosition(fn, openPos, syntax.MakePosition(&syntheticName, 1, len("result = ")+1))
}

```go
// captureLambdaAtPosition is a variant of captureLambda for synthesized
// lambdas (D4.1-01 interpolation desugarer). The caller supplies:
//   - fn: the *starlark.Function produced by ExecFileOptions
//   - userPos: the user's source position (for Pos field; error
//     attribution; lambda-ID input)
//   - bodyPos: the synthetic-file position where the lambda body lives
//     (for BodyPos field; D4-02 walker uses this to re-parse the
//     synthesized AST)
//
// captureLambda is unchanged — hand-written lambdas continue to flow
// through it with bodyPos == fn.Position() implicitly (zero BodyPos).
func (p *Parser) captureLambdaAtPosition(fn *starlark.Function, userPos, bodyPos syntax.Position) (*dag.CapturedLambda, error) {
    // Use the USER's file bytes for the ID hash — cosmetic edits to the
    // template string change the hash via the user file's content.
    fileBytes, ok := p.fileBytes[userPos.Filename()]
    if !ok || fileBytes == nil {
        return nil, &dag.ParseError{
            Pos: userPos,
            Msg: fmt.Sprintf("internal: file bytes for %q not cached", userPos.Filename()),
        }
    }
    id := dag.ComputeLambdaID(fileBytes, userPos)
    captured := &dag.CapturedLambda{
        ID:       id,
        Fn:       fn,
        Pos:      userPos,
        BodyPos:  bodyPos,
        FreeVars: starlark.StringDict{}, // synthesized lambdas have no free vars
    }
    captured.FreeVars.Freeze()
    p.lambdas[id] = captured
    return captured, nil
}
```
