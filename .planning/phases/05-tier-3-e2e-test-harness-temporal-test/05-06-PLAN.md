package testing

import (
    "crypto/sha256"
    "encoding/hex"
    "fmt"
    "io"

    "time"

// RunCLI is the non-*testing.T entry-point for pkg/cli. Mirrors pkgtesting.Run
// (Plan 04) but uses a synthetic Reporter (bareReporter) so the CLI can drive
// the harness without inventing a *testing.T. Returns the per-run pass/fail
// counts so pkg/cli/test.go can translate them to exit codes (D5-E4):
//
//   passed > 0, failed == 0     → exit 0
//   failed > 0                  → exit 1
//   err != nil                  → exit 1 (CLI prints err.Error() to stderr)
//
// Output (human or JSON per cfg.formatJSON) is written to cfg.formatOut.
//
// CLI-03 contract: default human format MUST NOT contain Go runtime stack
// frames. RunCLI relies on starlarktest.Error wrapping (which produces
// Starlark-callsite messages like "fail_test.star:2:5: assert.eq: ...")
// and uses bareReporter to format failures the SAME way runOneTest's
// subT.Errorf would, MINUS the Go-stack-trace machinery testing.T uses
// for its own Helper-marked frames.
//
// Plan 06 deviation D5-runner-cli-adapter: this duplicates ~80 LOC of
// runOneFile because Plan 05's runOneFile is *testing.T-bound. Refactor
// to a shared internal helper is acceptable in a future iteration; for
// v1 the duplication is contained and tested.
func RunCLI(dir string, opts ...Option) (passed, failed int, err error) {
    cfg := &runConfig{}
    for _, opt := range opts {
        if optErr := opt(cfg); optErr != nil {
            return 0, 0, optErr
        }
    }
    out := cfg.formatOut
    if out == nil {
        out = os.Stdout
    }
    var em *jsonEmitter
    if cfg.formatJSON {
        em = newJSONEmitter(out)
    }

    files, walkErr := DiscoverTestFiles(dir)
    if walkErr != nil {
        return 0, 0, fmt.Errorf("discover test files in %s: %w", dir, walkErr)
    }
    if len(files) == 0 {
        // Advisory + zero-counts. CLI translates (0, 0, nil) → exit 0 with
        // a stderr "no tests to run" line. Mirrors `go test ./...` semantics
        // when no test files are discoverable.
        if em == nil {
            fmt.Fprintf(out, "no *_test.star files under %s\n", dir)
        }
        return 0, 0, nil
    }

    runStart := time.Now()
    for _, file := range files {
        p, f := runOneFileCLI(file, cfg, out, em)
        passed += p
        failed += f
    }
    if em == nil {
        // Final summary line (D5-E1).
        if failed == 0 {
            fmt.Fprintf(out, "PASS  %d files  %d tests  (%.2fs)\n", len(files), passed+failed, time.Since(runStart).Seconds())
        } else {
            fmt.Fprintf(out, "FAIL  %d files  %d tests  %d failed  (%.2fs)\n", len(files), passed+failed, failed, time.Since(runStart).Seconds())
        }
    }
    return passed, failed, nil
}

// runOneFileCLI mirrors runner.go::runOneFile but uses bareReporter
// instead of *testing.T. Returns (passed, failed) for this file.
func runOneFileCLI(file string, cfg *runConfig, out io.Writer, em *jsonEmitter) (passed, failed int) {
    bytes, readErr := os.ReadFile(file)
    if readErr != nil {
        // Surface as a synthetic FAIL record so JSON consumers see it.
        fileBase := filepath.Base(file)
        if em != nil {
            em.emit("fail", fileBase, "", fmt.Sprintf("read %s: %v", file, readErr), 0)
        } else {
            fmt.Fprintf(out, "FAIL  %s  read error: %v\n", fileBase, readErr)
        }
        return 0, 1
    }

    reg := NewMockRegistry()
    ws := &WorkflowSpec{}
    var ctxRef *runContext
    builder := func(_ *parser.Parser, _ *starlark.Thread) starlark.Value {
        return newTesterModuleWithCtx(reg, ws, &ctxRef)
    }

    pOpts := []parser.Option{
        parser.WithTestMode(),
        parser.WithTestModule(builder),
        parser.WithExtensions(cfg.exts...),
    }
    p, parserErr := parser.NewParser(pOpts...)
    if parserErr != nil {
        // Same pattern as read error.
        fileBase := filepath.Base(file)
        if em != nil {
            em.emit("fail", fileBase, "", fmt.Sprintf("NewParser: %v", parserErr), 0)
        } else {
            fmt.Fprintf(out, "FAIL  %s  parser error: %v\n", fileBase, parserErr)
        }
        return 0, 1
    }
    if _, parseErr := p.ParseSource(file, bytes); parseErr != nil {
        fileBase := filepath.Base(file)
        // Render Starlark callsite — *dag.ParseError already includes
        // file:line:col.
        if em != nil {
            em.emit("fail", fileBase, "", parseErr.Error(), 0)
        } else {
            fmt.Fprintf(out, "FAIL  %s  parse error: %s\n", fileBase, parseErr.Error())
        }
        return 0, 1
    }

    // Build runContext (mirrors Plan 04 runner.go logic).
    flowsByName := map[string]*interpreter.ParsedFlow{}
    hashesByName := map[string]string{}
    fileBytesMap := p.FileBytes()
    for name, flow := range p.Flows() {
        var srcBytes []byte
        if b, ok := fileBytesMap[file]; ok {
            srcBytes = b
        }
        sum := sha256.Sum256(srcBytes)
        flowsByName[name] = &interpreter.ParsedFlow{
            Flow:    flow,
            Lambdas: p.Lambdas(),
        }
        hashesByName[name] = hex.EncodeToString(sum[:])
    }
    ctxRef = &runContext{flows: flowsByName, contentHashes: hashesByName}

    globals, ok := p.TestGlobals(file)
    if !ok {
        fileBase := filepath.Base(file)
        if em != nil {
            em.emit("fail", fileBase, "", "TestGlobals not captured", 0)
        } else {
            fmt.Fprintf(out, "FAIL  %s  TestGlobals not captured\n", fileBase)
        }
        return 0, 1
    }

    fileBase := filepath.Base(file)
    fileStem := strings.TrimSuffix(fileBase, ".star")
    fns := DiscoverTests(globals)

    var fileTotal, fileFailed int
    fileStart := time.Now()
    for _, fn := range fns {
        fullName := fileStem + "." + fn.Name
        if !MatchRunFilter(cfg.runRegex, fullName) {
            continue
        }
        fileTotal++

        if em != nil {
            em.emit("start", fileBase, fn.Name, "", 0)
            em.emit("run", fileBase, fn.Name, "", 0)
        }

        testStart := time.Now()
        rep := &bareReporter{}
        runOneTestCLI(rep, fn.Fn, reg, ws)
        elapsed := time.Since(testStart).Seconds()

        if rep.failed {
            fileFailed++
            if em != nil {
                em.emit("output", fileBase, fn.Name, rep.allMessages(), elapsed)
                em.emit("fail", fileBase, fn.Name, "", elapsed)
            } else {
                fmt.Fprint(out, formatHumanLine("FAIL", fn.Name, elapsed))
                // Indented assertion detail under the FAIL line (D5-E1
                // + D5-F3). Each captured Reporter.Error message is a
                // Starlark-callsite-aware string; indent two spaces.
                for _, line := range strings.Split(strings.TrimRight(rep.allMessages(), "\n"), "\n") {
                    if line == "" { continue }
                    fmt.Fprintf(out, "    %s\n", line)
                }
            }
        } else {
            if em != nil {
                em.emit("pass", fileBase, fn.Name, "", elapsed)
            } else {
                fmt.Fprint(out, formatHumanLine("PASS", fn.Name, elapsed))
            }
        }
    }

    // Per-file summary footer (human only, D5-E1).
    if em == nil && fileTotal > 0 {
        fileElapsed := time.Since(fileStart).Seconds()
        if fileFailed == 0 {
            fmt.Fprintf(out, "PASS  %s  %d tests  (%.2fs)\n", fileBase, fileTotal, fileElapsed)
        } else {
            fmt.Fprintf(out, "FAIL  %s  %d tests  %d failed  (%.2fs)\n", fileBase, fileTotal, fileFailed, fileElapsed)
        }
    }

    return fileTotal - fileFailed, fileFailed
}

// runOneTestCLI mirrors reporter.go::runOneTest but binds bareReporter
// (not *testing.T) to the starlarktest reporter slot. Same behavior:
// fresh thread per test, mock registry frame push/pop, Starlark.Call.
func runOneTestCLI(rep *bareReporter, fn *starlark.Function, reg *MockRegistry, ws *WorkflowSpec) {
    thread := &starlark.Thread{
        Name: fmt.Sprintf("test:%s:%s", fn.Position().Filename(), fn.Name()),
    }
    // starlarktest.SetReporter requires the Reporter interface; bareReporter
    // satisfies it via Error(args ...any).
    setStarlarkTestReporter(thread, rep)

    reg.PushTestFrame()
    defer reg.PopTestFrame()

    if _, err := starlark.Call(thread, fn, nil, nil); err != nil {
        // Starlark *EvalError already includes file:line:col.
        // Forward as one more Reporter.Error so callers see the same
        // failure shape regardless of whether the test died from
        // assert.* or a non-assertion eval error.
        rep.Error(err.Error())
    }
}

// bareReporter satisfies starlarktest.Reporter
// (`interface{ Error(args ...any) }`) without depending on *testing.T.
// Records every Error call so the CLI can render indented detail under
// the FAIL line (D5-E1).
type bareReporter struct {
    failed   bool
    messages []string
}

func (b *bareReporter) Error(args ...any) {
    b.failed = true
    b.messages = append(b.messages, fmt.Sprint(args...))
}

func (b *bareReporter) allMessages() string {
    return strings.Join(b.messages, "\n")
}

// setStarlarkTestReporter wraps starlarktest.SetReporter. Extracted so
// the real call can be swapped (e.g., in tests) without touching the
// runner. Implementation: just calls starlarktest.SetReporter(thread, rep).
//
// Note: starlarktest.SetReporter's signature is
// `func SetReporter(thread *starlark.Thread, r Reporter)`; bareReporter
// satisfies Reporter directly via its Error method.
func setStarlarkTestReporter(thread *starlark.Thread, rep *bareReporter) {
    // Imported via starlarktest.SetReporter; direct call.
    // (Actual import added when this file is implemented.)
    starlarkTestSetReporter(thread, rep)

// newTestCommand returns the skytime test subcommand.
//
// CLI-03: discover *_test.star files in <dir>, run the Tier-3 harness,
// report pass/fail with Starlark-callsite errors. NO Go stack traces in
// default output (D5-E1, D5-F3, CLI-03 explicit).
//
// Flags:
//   --run <regex>      Filter tests by `<file_basename>.<test_name>` (D5-E3).
//   --format human|json  Output format. Default human (D5-E1); json mirrors
//                        cmd/test2json schema (D5-E2).
//
// Inherits persistent --debug from root (D4-19); when set, RunE writes
// the error chain (cfg.debug) to stderr after the human output.
//
// Exit code mapping (D5-E4):
//   passed  > 0, failed == 0  → exit 0 (RunE returns nil)
//   failed  > 0               → exit 1 (RunE returns errSilent)
//   err     != nil            → exit 1 (RunE prints err to stderr, returns errSilent)
//   bad arg count             → exit 2 (cobra ExactArgs(1) before RunE)
func newTestCommand(cfg *config) *cobra.Command {
    var runPattern string
    var format string

    cmd := &cobra.Command{
        Use:   "test <dir>",
        Short: "Run Tier-3 .star tests in <dir>",
        Long: "Discovers *_test.star files in <dir> recursively, runs each def test_*() " +
            "function under the Tier-3 harness (testsuite.TestWorkflowEnvironment + " +
            "Starlark mocks), and reports pass/fail. Errors point at Starlark callsites " +
            "(file:line:col); --debug surfaces the underlying Go cause chain.",
        Args: cobra.ExactArgs(1),
        RunE: func(cmd *cobra.Command, args []string) error {
            dir := args[0]

            opts := []testingpkg.Option{
                testingpkg.WithExtensions(cfg.exts...),
                testingpkg.WithOutput(cmd.OutOrStdout()),
            }
            if runPattern != "" {
                opts = append(opts, testingpkg.WithRunFilter(runPattern))
            }
            if format != "" && format != "human" {
                opts = append(opts, testingpkg.WithFormat(format))
            }

            passed, failed, err := testingpkg.RunCLI(dir, opts...)
            if err != nil {
                fmt.Fprintf(cmd.ErrOrStderr(), "skytime test: %s\n", err.Error())
                return errSilent
            }
            // D5-E4: exit 1 on any failure.
            if failed > 0 {
                if cfg.debug {
                    fmt.Fprintf(cmd.ErrOrStderr(), "skytime test: %d failed, %d passed\n", failed, passed)
                }
                return errSilent
            }
            return nil
        },
    }

    cmd.Flags().StringVar(&runPattern, "run", "",
        "Filter tests by Go-regex against `<file_basename>.<test_name>` (e.g. ^users_test\\.test_existing)")
    cmd.Flags().StringVar(&format, "format", "human",
        "Output format: human (default; --- PASS / --- FAIL lines) or json (cmd/test2json mirror)")

    return cmd
}
