```go
import (
    "fmt"
    "testing"

    "go.starlark.net/starlark"

// runOneTest invokes a single `def test_*()` function under
// subT (a *testing.T from t.Run). Per D5-F1:
//
//   - Constructs a fresh *starlark.Thread per test (Pitfall #1: no
//     thread reuse across calls).
//   - Calls starlarktest.SetReporter(thread, subT) so assert.*
//     failures route to subT.Error (D5-F1, TEST-05).
//   - Pushes a per-test frame on the MockRegistry on entry; pops on
//     exit (D5-A4).
//   - Invokes fn via starlark.Call with no args (def test_* must
//     accept () — validated at discovery time, Plan 05).
//   - Catches non-assertion eval errors and surfaces them via
//     subT.Errorf (assert.* failures already routed to subT.Error
//     via the reporter; do not duplicate).
//
// reg + ws are owned by the runner; the per-test frame is the layer
// where tester.mock_action calls inside this def test_*() land.
func runOneTest(subT *testing.T, fn *starlark.Function, reg *MockRegistry, ws *WorkflowSpec) {
    subT.Helper()

    thread := &starlark.Thread{
        Name: fmt.Sprintf("test:%s:%s", fn.Position().Filename(), fn.Name()),
    }
    starlarktest.SetReporter(thread, subT)

    reg.PushTestFrame()
    defer reg.PopTestFrame()

    _, err := starlark.Call(thread, fn, nil, nil)
    if err != nil {
        // Starlark *EvalError already includes file:line:col via
        // err.Error(). Plain Go errors fall back to the same path.
        subT.Errorf("%s: %v", fn.Name(), err)
    }
}
```

```go
// helperGetTestFn parses src in test mode and returns the
// def test_<name> function from the captured globals.
func helperGetTestFn(t *testing.T, src, filename, name string) *starlark.Function {
    t.Helper()
    reg := NewMockRegistry()
    ws := &WorkflowSpec{}
    p, err := parser.NewParser(
        parser.WithTestMode(),
        parser.WithTestModule(func(_ *parser.Parser, _ *starlark.Thread) starlark.Value {
            return NewTesterModule(reg, ws)
        }),
    )
    require.NoError(t, err)
    _, err = p.ParseSource(filename, []byte(src))
    require.NoError(t, err)
    g, ok := p.TestGlobals(filename)
    require.True(t, ok)
    fn, ok := g[name].(*starlark.Function)
    require.True(t, ok, "missing %s in globals", name)
    return fn
}

// capturingReporter records starlarktest.Reporter.Error calls so
// reporter_test can assert on the failure messages directly.
type capturingReporter struct{ calls [][]any }

func (c *capturingReporter) Error(args ...any) {
    c.calls = append(c.calls, args)
}

func (c *capturingReporter) AllArgs() string {
    var sb strings.Builder
    for _, call := range c.calls {
        sb.WriteString(fmt.Sprintln(call...))
    }
    return sb.String()
}

// driveWithReporter parses src + invokes fn under a custom reporter.
// Bypasses the t.Run subtest layer so the test file can assert on
// reporter.calls directly.
func driveWithReporter(t *testing.T, src, filename, name string, rep starlarktest.Reporter) {
    t.Helper()
    fn := helperGetTestFn(t, src, filename, name)
    reg := NewMockRegistry()
    thread := &starlark.Thread{Name: "test:" + name}
    starlarktest.SetReporter(thread, rep)
    reg.PushTestFrame()
    defer reg.PopTestFrame()
    _, err := starlark.Call(thread, fn, nil, nil)
    if err != nil {
        rep.Error(err.Error())
    }
}

// TestAssert_FailureSurfacesInSubtestT — VALIDATION.md per-task map cite.
func TestAssert_FailureSurfacesInSubtestT(t *testing.T) {
    src := "def test_failing():\n    assert.eq(\"octocat\", \"default-user\")\n"
    cap := &capturingReporter{}
    driveWithReporter(t, src, "users_test.star", "test_failing", cap)
    require.NotEmpty(t, cap.calls, "assert.eq mismatch should call Reporter.Error")
    msg := cap.AllArgs()
    assert.Contains(t, msg, "users_test.star")
    assert.Contains(t, msg, "octocat")
    assert.Contains(t, msg, "default-user")
}

func TestAssert_AccumulatesMultipleFailuresInSubtest(t *testing.T) {
    src := "def test_two_failures():\n    assert.eq(1, 2)\n    assert.eq(3, 4)\n"

// TestRunOneTest_SubtestIsolation: t_a fails, t_b passes; verify
// independent subtest outcomes via t.Run.
func TestRunOneTest_SubtestIsolation(t *testing.T) {
    srcA := "def test_a():\n    assert.eq(1, 2)\n"
    srcB := "def test_b():\n    assert.eq(1, 1)\n"

            failedB = innerT.Failed()
```

```go
        // Detect by scanning the call stack for a non-builtin Function
        // frame upstream of the tester.run frame.
        if !isInsideDefTestStar(thread) {
            return nil, fmt.Errorf("tester.run must be called inside a def test_*() function (at %s)", callerPosFromThread(thread).String())
        }

        if len(args) != 0 {
            return nil, fmt.Errorf("tester.run: positional args not supported (use kwargs)")
        }
        var flowName string
        if err := starlark.UnpackArgs("tester.run", args, kwargs, "flow", &flowName); err != nil {
            return nil, err
        }
        if flowName == "" {
            return nil, fmt.Errorf("tester.run: flow must be non-empty")
        }

        ctx := *ctxRef
        if ctx == nil || ctx.parser == nil {
            return nil, fmt.Errorf("tester.run: runner context not initialized (was tester.run called outside pkg/testing.Run?)")
        }

        parsed, ok := ctx.flows[flowName]
        if !ok {
            return nil, fmt.Errorf("tester.run: flow %q not found in parser session", flowName)
        }
        hash, ok := ctx.contentHashes[flowName]
        if !ok {
            return nil, fmt.Errorf("tester.run: content_hash for flow %q not found", flowName)
        }

        initState := ws.InitState
        if initState == nil {
            initState = map[string]any{}
        }

        // D5-D1: always-on replay. Run twice with shared mock
        // registry + shared attempt counter.
        attempts := NewAttemptCounter()

        cap1, _, err1 := RunOnceCapturing(parsed, hash, initState, reg, attempts, nil)
        if err1 != nil {
            return nil, fmt.Errorf("tester.run (run 1): workflow error: %w", err1)
        }
        cap2, _, err2 := RunOnceCapturing(parsed, hash, initState, reg, attempts, nil)
        if err2 != nil {
            return nil, fmt.Errorf("tester.run (run 2): workflow error: %w", err2)
        }

        testCallsite := callerPosFromThread(thread)
        if d := FirstDivergentEvent(cap1, cap2, testCallsite); d != nil {
            // Report divergence via the active starlarktest reporter
            // (which the runner bound to the subtest's *testing.T).
            rep := starlarktest.GetReporter(thread)
            if rep != nil {
                rep.Error(d.Format())
            } else {
                // Defensive: no reporter wired (raw eval) — surface as
                // a Starlark error so the caller sees something.
                return nil, fmt.Errorf("%s", d.Format())
            }
        }

        return starlark.None, nil
    }
}

// isInsideDefTestStar walks the thread's call stack looking for a
// user-Function frame whose name starts with "test_". Returns true if
// found. Excludes builtin frames and the immediate tester.run frame
// itself.
func isInsideDefTestStar(thread *starlark.Thread) bool {
    cs := thread.CallStack()
    for _, frame := range cs {
        if frame.Name == "<builtin>" || frame.Name == "tester.run" || frame.Name == "tester.workflow" || frame.Name == "tester.mock_action" {
            continue
        }
        // The call stack uses the def function's name; def test_x ->
        // frame.Name == "test_x". Check the prefix.
        // Note: top-level <toplevel> frame has Name == "<toplevel>"
        // or similar; we want a NAMED function with test_ prefix.
        if frame.Name != "<toplevel>" && len(frame.Name) >= 5 && frame.Name[:5] == "test_" {
            return true
        }
    }
    return false
}
```

```go
type Option func(*runConfig) error

// WithExtensions registers extensions on the parser session. Mirrors
// parser.WithExtensions.
func WithExtensions(exts ...extension.Extension) Option {

// WithRunFilter sets a regex pattern (Go regexp) that filters tests
// by `<file_basename>.<test_name>`. Empty/absent = run all.
// Plan 05 ports the regex compile into the directory walker;
// Plan 04 accepts the option and stores it.
func WithRunFilter(pattern string) Option {

type runConfig struct {

// Run is the Go-level foundation API for Phase 5's test harness
// (Open Q7). pkg/cli/test.go (Plan 06) wraps this; Phase 6's example
// project can call it directly from a *_test.go in the consumer
// package.
//
// Behavior:
//   - Walks dir for *_test.star files. Plan 05 owns the recursive
//     walker; Plan 04 ships single-file support (panics if dir is

// + adds --format=json output records.)
func Run(t *testing.T, dir string, opts ...Option) {
    t.Helper()
    cfg := &runConfig{}
    for _, opt := range opts {
        if err := opt(cfg); err != nil {
            t.Fatalf("Run: option failed: %v", err)
        }
    }

    if err != nil {

    }

        t.Skipf("Run: no *_test.star files under %s", dir)
        return
    }

    for _, file := range files {
        runOneFile(t, file, cfg)
    }
}

// discoverTestFiles is Plan 04's stub. Plan 05 replaces with a
// recursive walk + .gitignore-style ignores.
func discoverTestFiles(dir string) ([]string, error) {
    info, err := os.Stat(dir)
    if err != nil {
        return nil, err
    }
    if !info.IsDir() {
        // Single-file path
        if !strings.HasSuffix(dir, "_test.star") {
            return nil, fmt.Errorf("path %s is not a *_test.star file", dir)
        }
        return []string{dir}, nil
    }
    entries, err := os.ReadDir(dir)
    if err != nil {
        return nil, err
    }
    var out []string
    for _, e := range entries {
        if e.IsDir() {
            continue
        }
        if strings.HasSuffix(e.Name(), "_test.star") {
            out = append(out, filepath.Join(dir, e.Name()))
        }
    }
    return out, nil
}

// runOneFile parses one *_test.star file and runs every discovered
// def test_*() under t.Run. The MockRegistry, WorkflowSpec, and
// runContext are shared across all tests in the file (file-frame
// mocks visible to every test; per-test frames push/pop inside
// runOneTest).
func runOneFile(t *testing.T, file string, cfg *runConfig) {
    t.Helper()

    bytes, err := os.ReadFile(file)
    if err != nil {
        t.Fatalf("read %s: %v", file, err)
    }

    reg := NewMockRegistry()
    ws := &WorkflowSpec{}
    var ctxRef *runContext
    builder := func(_ *parser.Parser, _ *starlark.Thread) starlark.Value {
        return newTesterModuleWithCtx(reg, ws, &ctxRef)
    }

    pOpts := []parser.Option{
        parser.WithTestMode(),
        parser.WithTestModule(builder),
        parser.WithExtensions(cfg.exts...),
    }
    p, err := parser.NewParser(pOpts...)
    if err != nil {
        t.Fatalf("NewParser: %v", err)
    }
    if _, err := p.ParseSource(file, bytes); err != nil {
        t.Fatalf("ParseSource %s: %v", file, err)
    }

    // Build runContext: parser flows + content hashes. The content
    // hash is sha256-hex of the file bytes registered to the flow.
    flowsByName := map[string]*interpreter.ParsedFlow{}
    hashesByName := map[string]string{}
    fileBytesMap := p.FileBytes()
    for name, flow := range p.Flows() {
        // Locate file bytes for this flow. Phase 1 stores them keyed
        // by absolute path. Plan 04 uses the test file's bytes
        // directly (single-file scope; load() across files is a
        // Phase 6 concern and out of scope for v1).
        var srcBytes []byte
        if b, ok := fileBytesMap[file]; ok {
            srcBytes = b
        }
        sum := sha256.Sum256(srcBytes)
        hash := hex.EncodeToString(sum[:])
        parsed := &interpreter.ParsedFlow{
            Flow:    flow,
            // Lambdas comes from the parser's Lambdas() accessor — same
            // shape as tests/differential_test.go's parseSrcAsFlow.
            Lambdas: p.Lambdas(),
        }
        flowsByName[name] = parsed
        hashesByName[name] = hash
    }
    ctxRef = &runContext{
        flows:         flowsByName,
        contentHashes: hashesByName,
    }

    // Discover def test_*() functions in the captured globals.
    globals, ok := p.TestGlobals(file)
    if !ok {
        t.Fatalf("TestGlobals not found for %s", file)
    }

    fileBase := filepath.Base(file)
    fileStem := strings.TrimSuffix(fileBase, ".star")  // e.g., "simple_test"

    var testNames []string
    for name := range globals {
        if strings.HasPrefix(name, "test_") {
            if fn, isFn := globals[name].(*starlark.Function); isFn && fn.NumParams() == 0 {
                testNames = append(testNames, name)
            }
        }
    }
    // Sort for replay determinism (Go map iteration is randomized).
    sortStrings(testNames)

    for _, name := range testNames {
        fn := globals[name].(*starlark.Function)
        if !matchesRunFilter(cfg.runPattern, fileStem+"."+name) {
            continue
        }
        t.Run(fileStem+"/"+name, func(subT *testing.T) {
            runOneTest(subT, fn, reg, ws)
        })
    }
    _ = dag.ActionRef{}  // touch dag package usage marker
}

// matchesRunFilter applies the Plan 05 regex semantics. Plan 04
// stub: empty pattern → match all; non-empty → strings.Contains.
// Plan 05 replaces with regexp.MatchString.
func matchesRunFilter(pattern, fullName string) bool {
    if pattern == "" {
        return true
    }
    return strings.Contains(fullName, pattern)
}

// sortStrings — stdlib sort import alias to keep the runner.go
// import block compact.
func sortStrings(s []string) {
    // Use sort.Strings; named function so the rest of the file reads
    // cleanly. (Avoids polluting the import block for a one-liner.)
    if len(s) < 2 { return }
    // Bubble — N is tiny (≤dozens of tests per file). Replace with
    // sort.Strings if any test file approaches >100 tests.
    for i := 0; i < len(s); i++ {
        for j := i + 1; j < len(s); j++ {
            if s[j] < s[i] {
                s[i], s[j] = s[j], s[i]
            }
        }
    }
}
```
