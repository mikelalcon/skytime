    // primitiveSet walks a flow's nodes (recursively into IfCond/ForEachParallel/CallFlow)
    // and returns the set of primitive kinds present.
    func primitiveSet(fl *dag.Flow) map[string]bool {
        out := map[string]bool{}
        var walk func(nodes []dag.Node)
        walk = func(nodes []dag.Node) {
            for _, n := range nodes {
                switch v := n.(type) {
                case *dag.Step:
                    // VERIFY field names against pkg/dag/node.go: Step has
                    // Action / Block / ActionFn / BlockFn (some may be Sets/exporters).
                    // Adjust the discriminator below to match the real field shapes.
                    switch {
                    case v.BlockFn != nil:
                        out["step_block_fn"] = true
                    case len(v.Block) > 0:
                        out["step_block"] = true
                    default:
                        out["step_seq"] = true
                    }
                case *dag.IfCond:
                    out["if_cond"] = true
                    walk(v.Then)
                    walk(v.Else)
                case *dag.Script:
                    out["script"] = true
                case *dag.ForEachParallel:
                    out["for_each_parallel"] = true
                    walk(v.Steps)
                case *dag.CallFlow:
                    out["call_flow"] = true
                }
            }
        }
        walk(fl.Steps)
        return out
    }

    // mapKeys returns sorted keys of a map (for stable error output).
    func mapKeys(m map[string]*dag.Flow) []string {
        keys := make([]string, 0, len(m))
        for k := range m { keys = append(keys, k) }
        return keys
    }
