package main

import (
    "context"
    "fmt"
    "os"
    "os/signal"
    "syscall"

    "github.com/mikelalcon/skytime/pkg/cli"
    "github.com/mikelalcon/skytime/pkg/extension"
    skyhttp "github.com/mikelalcon/skytime/pkg/extension/builtin/http"
)

func main() {
    _ = defaultBuildID
    ctx, cancel := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
    defer cancel()

    root, err := cli.NewRootCommand(
        cli.WithExtensions(skyhttp.New()),
        cli.WithCredentialHandler(noopCredentialHandler{}),
    )
    if err != nil {
        fmt.Fprintln(os.Stderr, err)
        os.Exit(1)
    }
    if err := root.ExecuteContext(ctx); err != nil {
        cli.RenderRootError(os.Stderr, err)
        os.Exit(1)
    }
}

type noopCredentialHandler struct{}
func (noopCredentialHandler) Resolve(_ context.Context, id string) (extension.Credential, error) {
    return nil, fmt.Errorf("no credential resolver configured: ... (id=%q)", id)
}
var _ extension.CredentialHandler = noopCredentialHandler{}
```

From RESEARCH.md § "Open Question 1": env-var SKYTIME_CREDFILE_PATH is the recommended override knob (least-friction; reader sets via shell). The example wires this in main.go.

From RESEARCH.md § "Open Question 3": pkg/cli/test.go does NOT forward CredentialHandler to pkg/testing.RunCLI. This is INTENDED — Tier-3 mocks are hermetic. The example's main.go ONLY needs WithCredentialHandler for `extbin run` (not `extbin test`).

**Lazy-load pattern (RESEARCH-implicit, plan-explicit):** If the user's first invocation is `extbin run public_repo_check.star` against the public GitHub API, NO credentials are needed. The `noopCredentialHandler` from cmd/skytime works for that path. But for the second-stage walkthrough (authenticated flows), the resolver MUST be the credfile.Resolver. The example handles both cleanly: wraps credfile.New() in a `lazyCredentialHandler` that only invokes credfile.New() on the FIRST call to Resolve(). If the user has no `~/.skytime-credentials` and runs only public flows, credfile.New() is never called, so the missing-file error never appears.

    // extbin is the custom Skytime CLI binary for the http-github-webhook
    // example. It demonstrates the canonical "build your own binary" pattern
    // (docs/cli-binary.md) by wiring three extensions (HTTP + GitHub +
    // Webhook) and a credfile-backed credential resolver into the same
    // pkg/cli root command tree that cmd/skytime uses.
    //
    // Subcommands inherited from pkg/cli (no custom additions):
    //   extbin validate <file.star>      static validation (Tier 1)
    //   extbin run <file.star> ...       trigger a workflow against a Temporal cluster
    //   extbin dev-server                spawn a local Temporal dev server
    //   extbin test <dir>                discover + run *_test.star (Tier 3)
    //
    // Credentials live at $HOME/.skytime-credentials (TOML; see
    // examples/http-github-webhook/.skytime-credentials.example for the
    // schema). Override via SKYTIME_CREDFILE_PATH if the file lives elsewhere.
    package main

    import (
        "context"
        "fmt"
        "os"
        "os/signal"
        "sync"
        "syscall"

        "github.com/mikelalcon/skytime/pkg/cli"
        "github.com/mikelalcon/skytime/pkg/extension"
        skyhttp "github.com/mikelalcon/skytime/pkg/extension/builtin/http"
        "github.com/mikelalcon/skytime/pkg/extension/credfile"

        skygh "github.com/mikelalcon/skytime/examples/http-github-webhook/extensions/github"
        skyweb "github.com/mikelalcon/skytime/examples/http-github-webhook/extensions/webhook"
    )

    func main() {
        ctx, cancel := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
        defer cancel()

        root, err := cli.NewRootCommand(
            cli.WithExtensions(
                skyhttp.New(),
                skygh.New(),
                skyweb.New(),
            ),
            cli.WithCredentialHandler(newLazyCredfileHandler(os.Getenv("SKYTIME_CREDFILE_PATH"))),
        )
        if err != nil {
            fmt.Fprintln(os.Stderr, err)
            os.Exit(1)
        }

        if err := root.ExecuteContext(ctx); err != nil {
            cli.RenderRootError(os.Stderr, err)
            os.Exit(1)
        }
    }

    // lazyCredfileHandler defers credfile.New() to the first Resolve() call.
    //
    // Why: the headline demo (`extbin run public_repo_check.star`) uses the
    // public GitHub API and never resolves any credential. If the reader
    // hasn't created ~/.skytime-credentials yet, eager construction would
    // fail at startup with a confusing "stat: no such file" before any
    // command runs. Lazy construction means that startup ALWAYS succeeds;
    // the credfile is read only when a flow actually references a
    // credential — at which point the user is expecting credfile to be
    // configured.
    //
    // Concurrency: pkg/activity/execute_batch.go invokes Resolve() from
    // multiple goroutines per heartbeat boundary. sync.Once + an atomic
    // ref protect the lazy initialization.
    type lazyCredfileHandler struct {
        path string

        once sync.Once
        once_inner *credfile.Resolver
        once_err   error
    }

    func newLazyCredfileHandler(pathOverride string) *lazyCredfileHandler {
        return &lazyCredfileHandler{path: pathOverride}
    }

    // Resolve implements extension.CredentialHandler. The first call
    // constructs the underlying credfile.Resolver; subsequent calls reuse it.
    // Construction errors (missing file, parse error, world-readable +
    // strict) are cached and returned on every subsequent call so the
    // surfaced message is stable.
    func (h *lazyCredfileHandler) Resolve(ctx context.Context, id string) (extension.Credential, error) {
        h.once.Do(func() {
            opts := []credfile.Option{}
            if h.path != "" {
                opts = append(opts, credfile.WithPath(h.path))
            }
            r, err := credfile.New(opts...)
            h.once_inner = r
            h.once_err = err
        })
        if h.once_err != nil {
            return nil, fmt.Errorf("extbin: load credfile (set SKYTIME_CREDFILE_PATH or copy .skytime-credentials.example to ~/.skytime-credentials): %w", h.once_err)
        }
        return h.once_inner.Resolve(ctx, id)
    }

    // Compile-time interface check.
    var _ extension.CredentialHandler = (*lazyCredfileHandler)(nil)
    ```

    Notes:
    - **No `defaultBuildID` indirection** — extbin doesn't ship release builds; the example is built by the reader from source. Skipping that variable keeps the file shorter and avoids a build_id.go sibling.
    - **Field naming `once_inner` / `once_err`**: deliberately ugly to flag they're sync.Once-protected; do not rename without acknowledging the race risk. (This pattern matches Phase 3's `clientDialFunc` lazy-resolver in `pkg/worker/boot.go` if the executor wants a parallel reference; otherwise the comments here are sufficient.)
    - The error-wrapping message at the surface ("set SKYTIME_CREDFILE_PATH or copy .skytime-credentials.example...") is the user-facing diagnostic. It MUST mention both the env var and the example file by name so a confused reader can self-recover without reading source.
    - **Why not just use a direct credfile.New() at startup with `WithStrictMode(false)`?** Because file-MISSING is different from file-INSECURE-MODE. credfile.New() returns an error on missing-file regardless of strict mode (the file MUST exist or the resolver has nothing to serve). Lazy-loading is the right escape valve for the "headline demo needs no credentials" path.

    import (
        "context"
        "errors"
        "os"
        "os/exec"
        "path/filepath"

        "testing"

        "github.com/stretchr/testify/assert"

    // public_repo_check.star path must not fail at startup if the user

    func TestLazyCredfileHandler_NoEagerConstruction(t *testing.T) {
        h := newLazyCredfileHandler("/no/such/credfile/exists/at/all")
        // No call to Resolve() — Once must not have fired.
        // We can't directly test once.Do bypass without reflection, but:
        require.NotNil(t, h)
        assert.Equal(t, "/no/such/credfile/exists/at/all", h.path)

    // user-recovery hint is present on the error message and the error
    // is cached (idempotent across multiple Resolve calls).
    func TestLazyCredfileHandler_FirstResolveSurfacesMissingFile(t *testing.T) {
        h := newLazyCredfileHandler("/no/such/credfile/exists")
        ctx := context.Background()

        _, err1 := h.Resolve(ctx, "x")
        require.Error(t, err1)
        assert.Contains(t, err1.Error(), "credfile")
        assert.Contains(t, err1.Error(), "SKYTIME_CREDFILE_PATH")

        assert.Equal(t, err1.Error(), err2.Error())

    // credfile DOES exist and contains a valid bearer entry, Resolve
    // returns the BearerCredential.
    func TestLazyCredfileHandler_HappyPathWithRealFile(t *testing.T) {
        dir := t.TempDir()
        path := filepath.Join(dir, ".skytime-credentials")
        require.NoError(t, os.WriteFile(path, []byte(`
[credentials.github_token]
type  = "bearer"
token = "test-pat"
`), 0o600))

        h := newLazyCredfileHandler(path)
        cred, err := h.Resolve(context.Background(), "github_token")
        require.NoError(t, err)
        bearer, ok := cred.(*extension.BearerCredential)
        require.True(t, ok)
        assert.Equal(t, "github_token", bearer.ID())
        assert.Equal(t, "test-pat", bearer.Token.Reveal())

        _, err = h.Resolve(context.Background(), "missing")
        require.Error(t, err)
        assert.True(t, errors.Is(err, extension.ErrUnknownCredential))
