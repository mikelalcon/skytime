    Step 3 — Implement `builtinResult` in `pkg/parser/builtins.go`:
    ```go
    func (p *Parser) builtinResult(thread *starlark.Thread, fn *starlark.Builtin, args starlark.Tuple, kwargs []starlark.Tuple) (starlark.Value, error) {
        // Reject positional args
        if len(args) > 0 {
            return nil, p.wrapBuiltinError("result", thread, fmt.Errorf("result: kwarg-only signature; use result(value={...})"))
        }
        var valueArg starlark.Value
        if err := starlark.UnpackArgs("result", args, kwargs, "value", &valueArg); err != nil {
            return nil, p.wrapBuiltinError("result", thread, err)
        }
        callPos := callerPosition(thread)

        // AST-level dict-literal recognition (D4.2-02). Re-parse the cached
        // file bytes at callPos and find the call expression whose Lparen
        // matches; inspect the value= argument's AST node.
        fileBytes, ok := p.fileBytes[callPos.Filename()]
        if !ok || fileBytes == nil {
            return nil, p.wrapBuiltinError("result", thread, fmt.Errorf("internal: file bytes for %q not cached", callPos.Filename()))
        }
        valueExpr, err := findResultValueArg(fileBytes, callPos.Filename(), callPos)
        if err != nil {
            return nil, p.wrapBuiltinError("result", thread, err)
        }
        dictExpr, isDict := valueExpr.(*syntax.DictExpr)
        if !isDict {
            return nil, p.wrapBuiltinError("result", thread, fmt.Errorf("result.value must be a dict literal"))
        }

        // Per-key capture + type inference.
        keys := make([]string, 0, len(dictExpr.List))
        values := make(map[string]*dag.CapturedLambda, len(dictExpr.List))
        types := make(map[string]any, len(dictExpr.List))   // any → parser.TypeInfo
        for _, entry := range dictExpr.List {
            de := entry.(*syntax.DictEntry)
            keyLit, ok := de.Key.(*syntax.Literal)
            if !ok || keyLit.Token != syntax.STRING {
                return nil, p.wrapBuiltinError("result", thread, fmt.Errorf("result.value: dict keys must be string literals"))
            }
            keyStr := keyLit.Value.(string)
            captured, capErr := p.captureResultValueLambda(de.Value, callPos, keyStr, fileBytes)
            if capErr != nil {
                return nil, p.wrapBuiltinError("result", thread, capErr)
            }
            keys = append(keys, keyStr)
            values[keyStr] = captured
            // Type inference: feed the user-source AST expression to inferType
            // against the CURRENT state schema. Plan 03's validator re-runs
            // inference against branch-fork state for the per-key match check;
            // the value here is a best-effort first pass useful for early
            // diagnostics but not authoritative until plan 03.
            //
            // For Wave-1, pass an empty schema (TypeOpaque for ctx.<x>); plan
            // 03's validator re-walks with the proper schema. Document this
            // limitation as a TODO comment + open question.
            types[keyStr] = inferType(de.Value, newStateSchema(), "ctx")
        }
        return wrapNode(&dag.Result{Pos: callPos, Keys: keys, Values: values, Types: types}), nil
    }

    // findResultValueArg re-parses fileBytes and locates the value= arg of
    // the result(...) call whose Lparen matches callPos. Returns the AST
    // expression node (NOT evaluated); caller asserts *syntax.DictExpr.
    // Mirror the implementation pattern from pkg/parser/block_fn_lint.go::
    // classifyBlockFn (re-parse + walk to find CallExpr by Lparen position).
    func findResultValueArg(fileBytes []byte, filename string, callPos syntax.Position) (syntax.Expr, error) { ... }
    ```
    The `findResultValueArg` helper mirrors `block_fn_lint.go::lookupOpSpec`'s re-parse approach: parse the file once via `(*syntax.FileOptions).Parse`, walk the resulting `*syntax.File` looking for `*syntax.CallExpr` whose `Lparen.Line == callPos.Line && Lparen.Col == callPos.Col`. From that CallExpr, find the kwarg whose Key.Name == "value" and return its Value AST node.
